using System;
using Xunit;
using Xunit.Abstractions;

namespace Smalltalk80.Tests;

/// <summary>
/// Tests to debug and fix file-out format parsing issues.
/// </summary>
public class FileOutParsingTests(ITestOutputHelper output)
{
    [Fact]
    public void FileOutFormat_SingleMethod_ParsesCorrectly()
    {
        ObjectMemory.Reset();

        var fileOutSource = @"
Object subclass: #TestClass
    instanceVariableNames: ''
    classVariableNames: ''
    poolDictionaries: ''
    category: 'Test-Debug'!

!TestClass methodsFor: 'testing'!
testMethod
    ^42
! !
";

        var tempFile = CreateTempFile(fileOutSource);

        try
        {
            var vm = new VirtualMachine();
            vm.LoadSystemFile(tempFile);

            var testClass = SystemDictionary.GetClass("TestClass");
            Assert.NotNull(testClass);

            var method = testClass.Lookup(new Symbol("testMethod"));
            Assert.NotNull(method);

            output.WriteLine("✅ Single method file-out parsing works");
        }
        finally
        {
            System.IO.File.Delete(tempFile);
        }
    }

    [Fact]
    public void FileOutFormat_MultipleMethodsInCategory_ParsesCorrectly()
    {
        ObjectMemory.Reset();

        var fileOutSource = @"
Object subclass: #TestClass
    instanceVariableNames: ''
    classVariableNames: ''
    poolDictionaries: ''
    category: 'Test-Debug'!

!TestClass methodsFor: 'testing'!
method1
    ^1
! !

!TestClass methodsFor: 'testing'!
method2
    ^2
! !
";

        var tempFile = CreateTempFile(fileOutSource);

        try
        {
            var vm = new VirtualMachine();
            vm.LoadSystemFile(tempFile);

            var testClass = SystemDictionary.GetClass("TestClass");
            Assert.NotNull(testClass);

            var method1 = testClass.Lookup(new Symbol("method1"));
            var method2 = testClass.Lookup(new Symbol("method2"));

            output.WriteLine($"Method1 found: {method1 != null}");
            output.WriteLine($"Method2 found: {method2 != null}");

            Assert.NotNull(method1);
            Assert.NotNull(method2);

            output.WriteLine("✅ Multiple methods in same category parsing works");
        }
        finally
        {
            System.IO.File.Delete(tempFile);
        }
    }

    [Fact]
    public void FileOutFormat_MultipleCategories_ParsesCorrectly()
    {
        ObjectMemory.Reset();

        var fileOutSource = @"
Object subclass: #TestClass
    instanceVariableNames: ''
    classVariableNames: ''
    poolDictionaries: ''
    category: 'Test-Debug'!

!TestClass methodsFor: 'category1'!
method1
    ^1
! !

!TestClass methodsFor: 'category2'!

Object subclass: #SimpleCounter
    instanceVariableNames: 'count'
    classVariableNames: ''
    poolDictionaries: ''
    category: 'Test-Integration'!

!SimpleCounter methodsFor: 'initialization'!
initialize
    count := 0.
    ^self
! !

!SimpleCounter methodsFor: 'accessing'!
getCount
    ^count
! !

!SimpleCounter methodsFor: 'incrementing'!
increment
    count := count + 1.
    ^count
! !
";

        // Load the complete class
        var vm = new VirtualMachine();
        var tempFile = CreateTempFile(classSource);
        vm.LoadSystemFile(tempFile);

        // Get the class and create an instance
        var counterClass = SystemDictionary.GetClass("SimpleCounter");
        var counterInstance = new SmalltalkObject(counterClass, counterClass.InstanceVariables.Count);

        var interpreter = new Interpreter();

        // Test initialization
        var initMethod = counterClass.Lookup(new Symbol("initialize"));
        Assert.NotNull(initMethod);
        interpreter.SetActiveContext(counterInstance, initMethod);
        var initResult = interpreter.ExecuteMethod();
        Assert.Equal(counterInstance, initResult);

        // Test getting initial count

        Assert.NotNull(getCountMethod);
        interpreter.SetActiveContext(counterInstance, getCountMethod);
        var countResult = interpreter.ExecuteMethod();
        Assert.Equal(0, countResult);

        // Test increment
        var incrementMethod = counterClass.Lookup(new Symbol("increment"));
        Assert.NotNull(incrementMethod);
        interpreter.SetActiveContext(counterInstance, incrementMethod);
        var incrementResult = interpreter.ExecuteMethod();
        Assert.Equal(1, incrementResult);

        // Verify count was updated
        interpreter.SetActiveContext(counterInstance, getCountMethod);
        var finalCount = interpreter.ExecuteMethod();
        Assert.Equal(1, finalCount);

        output.WriteLine("✅ Full class workflow executed successfully");

        File.Delete(tempFile);
    }

    {
        // Ensure standard library is loaded by creating a VirtualMachine instance
        var vm = new VirtualMachine();

        // Create a minimal receiver if none provided
