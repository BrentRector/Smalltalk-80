namespace Smalltalk80.Parser;

/// <summary>
/// Base type for all AST nodes produced by the parser.
/// </summary>
public abstract record AstNode;

/// <summary>
/// Root AST node representing a parsed script.
/// </summary>
/// <param name="Sequence">The top-level sequence.</param>
public sealed record ScriptNode(SequenceNode Sequence) : AstNode;

/// <summary>
/// A sequence of statements with optional temporary variables.
/// </summary>
/// <param name="Temporaries">Temporary variable names.</param>
/// <param name="Statements">The contained statements.</param>
public sealed record SequenceNode(IReadOnlyList<string> Temporaries, IReadOnlyList<StatementNode> Statements) : AstNode;

/// <summary>
/// Base type for all statement nodes.
/// </summary>
public abstract record StatementNode : AstNode;

/// <summary>
/// Statement consisting of a single expression.
/// </summary>
/// <param name="Expression">The expression to evaluate.</param>
public sealed record ExpressionStatement(ExpressionNode Expression) : StatementNode;

/// <summary>
/// Statement that returns the value of an expression.
/// </summary>
/// <param name="Expression">The expression whose value is returned.</param>
public sealed record AnswerStatement(ExpressionNode Expression) : StatementNode;

/// <summary>
/// Statement representing a primitive declaration within a sequence.
/// </summary>
/// <param name="PrimitiveIndex">The index of the primitive to invoke.</param>
public sealed record PrimitiveStatement(int PrimitiveIndex) : StatementNode;

/// <summary>
/// Base type for expression nodes.
/// </summary>
public abstract record ExpressionNode : AstNode;

/// <summary>
/// A literal value such as an integer or string.
/// </summary>
/// <param name="Value">The literal value.</param>
public sealed record LiteralExpression(object Value) : ExpressionNode;

/// <summary>
/// Reference to a variable in the current scope.
/// </summary>
/// <param name="Name">The variable name.</param>
public sealed record VariableExpression(string Name) : ExpressionNode;

/// <summary>
/// Assignment of a value to a variable.
/// </summary>
/// <param name="Variable">The variable being assigned.</param>
/// <param name="Value">The assigned value.</param>
public sealed record AssignmentExpression(string Variable, ExpressionNode Value) : ExpressionNode;

/// <summary>
/// A binary message send such as <c>a + b</c>.
/// </summary>
/// <param name="Receiver">The receiver expression.</param>
/// <param name="Selector">The binary selector.</param>
/// <param name="Argument">The argument expression.</param>
public sealed record BinarySendExpression(ExpressionNode Receiver, string Selector, ExpressionNode Argument) : ExpressionNode;

/// <summary>
/// A keyword message send such as <c>a at: 1</c>.
/// </summary>
/// <param name="Receiver">The receiver expression.</param>
/// <param name="SelectorParts">The individual keyword parts including trailing colons.</param>
/// <param name="Arguments">The argument expressions corresponding to <paramref name="SelectorParts"/>.</param>
public sealed record KeywordSendExpression(
    ExpressionNode Receiver,
    IReadOnlyList<string> SelectorParts,
    IReadOnlyList<ExpressionNode> Arguments) : ExpressionNode;

/// <summary>
/// A unary message send such as <c>a size</c>.
/// </summary>
/// <param name="Receiver">The receiver expression.</param>
/// <param name="Selector">The unary selector.</param>
public sealed record UnarySendExpression(ExpressionNode Receiver, string Selector) : ExpressionNode;

/// <summary>
/// A cascade of messages sent to the same receiver.
/// </summary>
/// <param name="Receiver">The common receiver expression.</param>
/// <param name="Messages">The messages sent in order.</param>
public sealed record CascadeExpression(ExpressionNode Receiver, IReadOnlyList<ExpressionNode> Messages) : ExpressionNode;

/// <summary>
/// Represents one of the Smalltalk pseudo variables such as <c>self</c> or <c>nil</c>.
/// </summary>
/// <param name="Name">The pseudo variable name.</param>
public sealed record PseudoVariableExpression(string Name) : ExpressionNode;

/// <summary>
/// Expression that causes an immediate return from the enclosing method with the value of
/// <paramref name="Value"/>.
/// </summary>
/// <param name="Value">The expression whose result is returned.</param>
public sealed record ReturnExpression(ExpressionNode Value) : ExpressionNode;

/// <summary>
/// Root node representing a file-in consisting of multiple chunks.
/// </summary>
/// <param name="Chunks">The ordered chunks in the file.</param>
public sealed record FileNode(IReadOnlyList<AstNode> Chunks) : AstNode;

/// <summary>
/// Node representing a top-level expression chunk within a file-in source.
/// </summary>
/// <param name="Expression">The expression evaluated by the chunk.</param>
public sealed record ExpressionChunkNode(ExpressionNode Expression) : AstNode;

/// <summary>
/// Node representing a class definition chunk.
/// </summary>
/// <param name="Superclass">The superclass name.</param>
/// <param name="ClassName">The class being defined.</param>
/// <param name="InstanceVariables">Declared instance variables for the class.</param>
/// <param name="ClassVariables">Declared class variables shared by all instances.</param>
/// <param name="PoolDictionaries">Names of any pool dictionaries referenced by the class.</param>
/// <param name="Category">Organizational category for the class.</param>
public sealed record ClassDefinitionNode(
    string Superclass,
    string ClassName,
    IReadOnlyList<string> InstanceVariables,
    IReadOnlyList<string> ClassVariables,
    IReadOnlyList<string> PoolDictionaries,
    string? Category) : AstNode;

/// <summary>
/// Node representing a class comment chunk.
/// </summary>
/// <param name="ClassName">The class being documented.</param>
/// <param name="Comment">The text of the comment.</param>
public sealed record ClassCommentNode(string ClassName, string Comment) : AstNode;

/// <summary>
/// Node representing a method category header chunk.
/// </summary>
/// <param name="ClassName">The class to which subsequent methods belong.</param>
/// <param name="IsClassSide">Whether the methods belong to the metaclass.</param>
/// <param name="Category">The category name.</param>
public sealed record MethodHeaderNode(string ClassName, bool IsClassSide, string Category) : AstNode;

/// <summary>
/// Node representing a method category chunk containing a header and multiple method definitions.
/// </summary>
/// <param name="Header">The method category header.</param>
/// <param name="Methods">The method definitions in this category.</param>
public sealed record MethodCategoryNode(MethodHeaderNode Header, IReadOnlyList<MethodDefinitionNode> Methods) : AstNode;

/// <summary>
/// Node representing a method definition.
/// </summary>
/// <param name="Selector">The method selector.</param>
/// <param name="Arguments">Names of the method parameters in declaration order.</param>
/// <param name="Body">The method body.</param>
/// <param name="Comment">Optional method comment text.</param>
/// <param name="Source">The raw Smalltalk source for the method body.</param>
/// <param name="HasPrimitive">Whether the method invokes a primitive.</param>
/// <param name="PrimitiveIndex">The primitive index if <paramref name="HasPrimitive"/> is <c>true</c>.</param>
public sealed record MethodDefinitionNode(
    string Selector,
    IReadOnlyList<string> Arguments,
    SequenceNode Body,
    string? Comment,
    string Source = "",
    bool HasPrimitive = false,
    int PrimitiveIndex = 0) : AstNode;

/// <summary>
/// Expression representing a block literal.
/// </summary>
/// <param name="Parameters">Names of the block parameters in declaration order.</param>
/// <param name="Body">The sequence executed when the block is evaluated.</param>
public sealed record BlockExpression(IReadOnlyList<string> Parameters, SequenceNode Body) : ExpressionNode;

/// <summary>
/// Expression representing a dynamic array literal evaluated at runtime.
/// </summary>
/// <param name="Elements">The element expressions comprising the array.</param>
public sealed record DynamicArrayExpression(IReadOnlyList<ExpressionNode> Elements) : ExpressionNode;

/// <summary>
/// Expression representing a dynamic dictionary literal evaluated at runtime.
/// </summary>
/// <param name="Elements">The expressions contained within the dictionary literal.</param>
public sealed record DynamicDictionaryExpression(IReadOnlyList<ExpressionNode> Elements) : ExpressionNode;

// file: Smalltalk80/Parser/ClassDefBuildingVisitor.cs
using Antlr4.Runtime.Misc;
using Smalltalk80.Generated;

namespace Smalltalk80.Parser;

/// <summary>
/// Visitor that builds class definition and method AST nodes from the parse tree.
/// Extends the simple AST building visitor to handle file-in chunks.
/// </summary>
public class ClassDefBuildingVisitor : SimpleAstBuildingVisitor
{
    /// <summary>
    /// Visit a file-in context and create a FileNode containing all chunks.
    /// </summary>
    public override object? VisitFileIn(Smalltalk80Parser.FileInContext context)
    {
        var chunks = new List<AstNode>();

        foreach (var chunk in context.chunk())
        {
            if (Visit(chunk) is AstNode astNode)
            {
                chunks.Add(astNode);
            }
        }

        return new FileNode(chunks);
    }

    /// <summary>
    /// Visit a class definition chunk.
    /// </summary>
    public override object? VisitClassDefChunk(Smalltalk80Parser.ClassDefChunkContext context)
    {
        var header = context.classHeader();
        var body = context.classBody();

        // Extract superclass name (the first identifier)
        var superclass = header.className().GetText();

        // Extract class name from symbol (remove '#' prefix)
        var classSymbol = header.symbolConst().GetText();
        var className = classSymbol.Substring(1); // Remove '#' prefix

        // Extract instance variables, class variables, etc. from body
        var instanceVariables = new List<string>();
        var classVariables = new List<string>();
        var poolDictionaries = new List<string>();
        string? category = null;

        foreach (var slotSpec in body.classSlotSpec())
        {
            var keyword = slotSpec.identifierColon().GetText();
            var stringValue = slotSpec.stringConst().GetText();
            // Remove quotes from string
            var value = stringValue.Substring(1, stringValue.Length - 2);

            switch (keyword)
            {
                case "instanceVariableNames:":
                    instanceVariables.AddRange(ParseVariableNames(value));
                    break;
                case "classVariableNames:":
                    classVariables.AddRange(ParseVariableNames(value));
                    break;
                case "poolDictionaries:":
                    poolDictionaries.AddRange(ParseVariableNames(value));
                    break;
            }
        }

        // Extract category
        if (body.categoryDecl() != null)
        {
            var categoryString = body.categoryDecl().stringConst().GetText();
            category = categoryString.Substring(1, categoryString.Length - 2);
        }

        return new ClassDefinitionNode(
            superclass,
            className,
            instanceVariables,
            classVariables,
            poolDictionaries,
            category
        );
    }

    /// <summary>
    /// Visit a method category chunk.
    /// </summary>
    public override object? VisitMethodCategoryChunk(Smalltalk80Parser.MethodCategoryChunkContext context)
    {
        var className = context.className().GetText();
        var contextText = context.GetText();

        // More robust class side detection - check if "class" appears after the class name
        var isClassSide = contextText.Contains($"{className}class");

        var categoryString = context.stringConst().GetText();
        var category = categoryString.Substring(1, categoryString.Length - 2);

        var header = new MethodHeaderNode(className, isClassSide, category);

        // Process all method definitions in this category
        var methods = new List<MethodDefinitionNode>();
        foreach (var methodDefContext in context.methodDef())
        {
            if (Visit(methodDefContext) is MethodDefinitionNode methodDef)
            {
                methods.Add(methodDef);
            }
        }

        return new MethodCategoryNode(header, methods);
    }

    /// <summary>
    /// Visit a method definition.
    /// </summary>
    public override object? VisitMethodDef(Smalltalk80Parser.MethodDefContext context)
    {
        var header = context.methodHeader();
        var temporariesCtx = context.temporaries();
        var primitiveCtx = context.primitiveDecl();
        var body = context.methodBody();

        // Parse method selector and arguments
        var (selector, arguments) = ParseMethodHeader(header);

        // Parse temporaries
        var temporaries = new List<string>();
        if (temporariesCtx != null)
        {
            foreach (var temp in temporariesCtx.temporaryName())
            {
                temporaries.Add(temp.GetText());
            }
        }

        // Parse primitive declaration
        bool hasPrimitive = primitiveCtx != null;
        int primitiveIndex = 0;
        if (hasPrimitive && primitiveCtx != null)
        {
            var primitiveText = primitiveCtx.GetText();
            // Extract number from "<primitive: N>"
            var match = System.Text.RegularExpressions.Regex.Match(primitiveText, @"<primitive:\s*(\d+)>");
            if (match.Success)
            {
                primitiveIndex = int.Parse(match.Groups[1].Value);
            }
        }

        // Parse method body
        var statements = new List<StatementNode>();
        if (body.statements() != null)
        {
            var stmtList = (List<StatementNode>)Visit(body.statements())!;
            statements.AddRange(stmtList);
        }

        // Check for explicit return
        if (body.returnStatement() != null)
        {
            statements.Add((StatementNode)Visit(body.returnStatement())!);
        }

        var sequence = new SequenceNode(temporaries, statements);

        // Extract just the method body source (not the header), and remove trailing BANG
        var methodBodySource = body?.GetText() ?? "";
        if (methodBodySource.EndsWith("!"))
        {
            methodBodySource = methodBodySource.Substring(0, methodBodySource.Length - 1);
        }

        return new MethodDefinitionNode(
            selector,
            arguments,
            sequence,
            null, // comment - TODO: Extract from comments
            methodBodySource, // source - just the body without BANG
            hasPrimitive,
            primitiveIndex
        );
    }

    /// <summary>
    /// Visit a class comment chunk.
    /// </summary>
    public override object? VisitClassCommentChunk(Smalltalk80Parser.ClassCommentChunkContext context)
    {
        var className = context.className().GetText();
        var commentString = context.stringConst().GetText();
        var comment = commentString.Substring(1, commentString.Length - 2);

        return new ClassCommentNode(className, comment);
    }

    /// <summary>
    /// Visit an expression chunk.
    /// </summary>
    public override object? VisitExpressionChunk(Smalltalk80Parser.ExpressionChunkContext context)
    {
        var expression = (ExpressionNode)Visit(context.expression())!;
        return new ExpressionChunkNode(expression);
    }

    /// <summary>
    /// Parse method header to extract selector and argument names.
    /// </summary>
    private (string selector, List<string> arguments) ParseMethodHeader(Smalltalk80Parser.MethodHeaderContext header)
    {
        var arguments = new List<string>();

        // Check for unary selector
        if (header.unarySelector() != null)
        {
            return (header.unarySelector().GetText(), arguments);
        }

        // Check for binary selector
        if (header.binarySelector() != null && header.formalArg() != null && header.formalArg().Length > 0)
        {
            arguments.Add(header.formalArg()[0].GetText());
            return (header.binarySelector().GetText(), arguments);
        }

        // Check for keyword selector
        if (header.identifierColon() != null)
        {
            var selectorParts = new List<string>();
            var formalArgs = header.formalArg();

            // Debug the raw text to see what's being parsed
            // Console.WriteLine($"[DEBUG_PARSER] Method header raw text: '{header.GetText()}'");
            // Console.WriteLine($"[DEBUG_PARSER] Found {header.identifierColon().Length} keyword parts");
            // Console.WriteLine($"[DEBUG_PARSER] Found {formalArgs.Length} formal args");

            for (int i = 0; i < header.identifierColon().Length; i++)
            {
                var keywordPart = header.identifierColon(i).GetText();
                // Console.WriteLine($"[DEBUG_PARSER] Keyword part {i}: '{keywordPart}'");
                selectorParts.Add(keywordPart);

                if (i < formalArgs.Length)
                {
                    var argName = formalArgs[i].GetText();
                    // Console.WriteLine($"[DEBUG_PARSER] Arg {i}: '{argName}'");
                    arguments.Add(argName);
                }
            }

            var selector = string.Join("", selectorParts);
            // Console.WriteLine($"[DEBUG_PARSER] Final selector: '{selector}'");
            return (selector, arguments);
        }

        throw new InvalidOperationException("Unable to parse method header");
    }

    /// <summary>
    /// Parse variable names from a space-separated string.
    /// </summary>
    private static List<string> ParseVariableNames(string variableString)
    {
        if (string.IsNullOrWhiteSpace(variableString))
            return new List<string>();

        return variableString.Split(new[] { ' ', '\t', '\n', '\r' },
            StringSplitOptions.RemoveEmptyEntries).ToList();
    }
}

// file: Smalltalk80/Parser/FileInReader.cs
using System;
using System.Collections.Generic;
using System.IO;
using Antlr4.Runtime;
using Smalltalk80.Generated;

namespace Smalltalk80.Parser;

/// <summary>
/// Provides parsing utilities for classic Smalltalk file-in sources.
/// </summary>
public static class FileInReader
{
    /// <summary>
    /// Parses a file-in source into an abstract syntax tree using the ANTLR
    /// grammar. Method bodies are extracted verbatim for later compilation.
    /// </summary>
    /// <param name="path">Path of the Smalltalk file to parse.</param>
    /// <returns>A <see cref="FileNode"/> representing the file contents.</returns>
    public static FileNode Parse(string path)
    {
        var source = File.ReadAllText(path);
        var input = new AntlrInputStream(source);
        var lexer = new Smalltalk80Lexer(input);
        var tokens = new CommonTokenStream(lexer);
        var parser = new Smalltalk80Parser(tokens);
        parser.Interpreter.PredictionMode = Antlr4.Runtime.Atn.PredictionMode.LL_EXACT_AMBIG_DETECTION;
        parser.RemoveErrorListeners();
        var fileCtx = parser.fileIn();
        var visitor = new ClassDefBuildingVisitor();

        return (FileNode)visitor.VisitFileIn(fileCtx)!;
    }
}

// file: Smalltalk80/Parser/SimpleAstBuildingVisitor.cs
using System.Linq;
using Antlr4.Runtime.Misc;
using Antlr4.Runtime.Tree;
using Smalltalk80.Generated;

namespace Smalltalk80.Parser;

/// <summary>
/// Simple AST building visitor that matches the current grammar structure.
/// </summary>
public class SimpleAstBuildingVisitor : Smalltalk80BaseVisitor<object?>
{
    /// <summary>
    /// Visit a return statement context and create an AnswerStatement.
    /// </summary>
    public override object? VisitReturnStatement(Smalltalk80Parser.ReturnStatementContext context)
    {
        var expression = (ExpressionNode)Visit(context.expression())!;
        return new AnswerStatement(expression);
    }

    /// <summary>
    /// Visit an assignment expression context.
    /// </summary>
    public override object? VisitAssignmentExpr(Smalltalk80Parser.AssignmentExprContext context)
    {
        // Check if this is an assignment (variable ASSIGN assignmentExpr)
        if (context.variable() != null && context.ASSIGN() != null && context.assignmentExpr() != null)
        {
            var variable = context.variable().GetText();
            var value = (ExpressionNode)Visit(context.assignmentExpr())!;
            return new AssignmentExpression(variable, value);
        }

        // Otherwise, it's just a cascade expression
        return Visit(context.cascadeExpr());
    }

    /// <summary>
    /// Visit an expression statement context.
    /// </summary>
    public override object? VisitExpressionStatement(Smalltalk80Parser.ExpressionStatementContext context)
    {
        var expression = (ExpressionNode)Visit(context.expression())!;
        return new ExpressionStatement(expression);
    }

    /// <summary>
    /// Visit an expression context.
    /// </summary>
    public override object? VisitExpression(Smalltalk80Parser.ExpressionContext context)
    {
        return Visit(context.assignmentExpr());
    }

    /// <summary>
    /// Visit a cascade expression context.
    /// </summary>
    public override object? VisitCascadeExpr(Smalltalk80Parser.CascadeExprContext context)
    {
        try
        {
            var messageExprContext = context.messageExpr();
            if (messageExprContext == null)
            {
                var lineInfo = context.Start != null ? $"Line {context.Start.Line}, Column {context.Start.Column}" : "Unknown location";
                var contextText = context.GetText();
                throw new InvalidOperationException($"Cascade expression has no base message expression at {lineInfo}. Context: '{contextText}'");
            }

            var baseExprResult = Visit(messageExprContext);
            if (baseExprResult == null)
            {
                throw new InvalidOperationException("Cascade base expression returned null");
            }
            var baseExpr = (ExpressionNode)baseExprResult;

            var messageParts = context.messagePart();
            if (messageParts.Length == 0)
            {
                return baseExpr;
            }

            // Create a cascade expression
            var messages = new List<ExpressionNode>();
            foreach (var part in messageParts)
            {
                var partResult = Visit(part);
                if (partResult != null)
                {
                    messages.Add((ExpressionNode)partResult);
                }
            }

            return new CascadeExpression(baseExpr, messages);
        }
        catch (Exception ex)
        {
            var lineInfo = context.Start != null ? $"Line {context.Start.Line}, Column {context.Start.Column}" : "Unknown location";
            var contextText = context.GetText();
            Console.WriteLine($"[PARSER_WARNING] Error in VisitCascadeExpr at {lineInfo}: {ex.Message}. Context: '{contextText}'. Returning placeholder expression to continue parsing.");
            // Return a placeholder literal to continue parsing instead of crashing
            return new LiteralExpression("<<PARSE_ERROR>>");
        }
    }

    /// <summary>
    /// Visit a message expression context.
    /// </summary>
    public override object? VisitMessageExpr(Smalltalk80Parser.MessageExprContext context)
    {
        try
        {
            var messageBase = context.messageBase();
            if (messageBase == null)
            {
                var lineInfo = context.Start != null ? $"Line {context.Start.Line}, Column {context.Start.Column}" : "Unknown location";
                var contextText = context.GetText();
                Console.WriteLine($"[PARSER_WARNING] MessageExpr has null messageBase at {lineInfo}. Context: '{contextText}'. Skipping this expression.");
                // Return a placeholder error expression to continue parsing
                return new LiteralExpression("<<PARSE_ERROR>>");
            }

            var receiverResult = Visit(messageBase);
            if (receiverResult == null)
            {
                throw new InvalidOperationException($"Message receiver Visit returned null for messageBase type: {messageBase.GetType().Name}");
            }
            var receiver = (ExpressionNode)receiverResult;

            // Apply all message sends to the receiver
            var current = receiver;
            for (int i = 0; i < context.ChildCount; i++)
            {
                var child = context.GetChild(i);
                if (child is Smalltalk80Parser.UnaryMessageContext unary)
                {
                    var selector = unary.unarySelector().GetText();
                    current = new UnarySendExpression(current, selector);
                }
                else if (child is Smalltalk80Parser.BinaryMessageContext binary)
                {
                    var selector = binary.binarySelector().GetText();
                    var argumentResult = Visit(binary.messageBase());
                    if (argumentResult == null)
                    {
                        throw new InvalidOperationException("Binary message argument cannot be null");
                    }
                    var argument = (ExpressionNode)argumentResult;
                    current = new BinarySendExpression(current, selector, argument);
                }
                else if (child is Smalltalk80Parser.KeywordMessageContext keyword)
                {
                    var parts = new List<string>();
                    var arguments = new List<ExpressionNode>();

                    foreach (var keywordPart in keyword.keywordPart())
                    {
                        parts.Add(keywordPart.IDENTIFIER().GetText() + ":");
                        var argResult = Visit(keywordPart.nonKeywordExpr());
                        if (argResult == null)
                        {
                            throw new InvalidOperationException("Keyword message argument cannot be null");
                        }
                        arguments.Add((ExpressionNode)argResult);
                    }

                    current = new KeywordSendExpression(current, parts, arguments);
                }
            }

            return current;
        }
        catch (Exception ex)
        {
            var lineInfo = context.Start != null ? $"Line {context.Start.Line}, Column {context.Start.Column}" : "Unknown location";
            var contextText = context.GetText();
            throw new InvalidOperationException($"Error in VisitMessageExpr at {lineInfo}: {ex.Message}. Context: '{contextText}'", ex);
        }
    }

    /// <summary>
    /// Visit a message base context.
    /// </summary>
    public override object? VisitMessageBase(Smalltalk80Parser.MessageBaseContext context)
    {
        if (context.parenthesizedExpr() != null)
            return Visit(context.parenthesizedExpr());
        if (context.block() != null)
            return Visit(context.block());
        if (context.literal() != null)
            return Visit(context.literal());
        if (context.variable() != null)
            return Visit(context.variable());
        if (context.negativeNumber() != null)
            return Visit(context.negativeNumber());

        throw new InvalidOperationException("Unknown message base type");
    }

    /// <summary>
    /// Visit a variable context.
    /// </summary>
    public override object? VisitVariable(Smalltalk80Parser.VariableContext context)
    {
        return new VariableExpression(context.GetText());
    }

    /// <summary>
    /// Visit a literal context.
    /// </summary>
    public override object? VisitLiteral(Smalltalk80Parser.LiteralContext context)
    {
        if (context.numberLiteral() != null)
            return Visit(context.numberLiteral());
        if (context.stringConst() != null)
            return Visit(context.stringConst());
        if (context.symbolConst() != null)
            return Visit(context.symbolConst());
        if (context.arrayLiteral() != null)
            return Visit(context.arrayLiteral());
        if (context.byteArrayLiteral() != null)
            return Visit(context.byteArrayLiteral());
        if (context.characterLiteral() != null)
            return Visit(context.characterLiteral());

        throw new InvalidOperationException("Unknown literal type");
    }

    /// <summary>
    /// Visit a number literal context.
    /// </summary>
    public override object? VisitNumberLiteral(Smalltalk80Parser.NumberLiteralContext context)
    {
        var text = context.GetText();
        if (text.Contains('.') || text.Contains('e') || text.Contains('E'))
        {
            return new LiteralExpression(double.Parse(text));
        }
        else if (text.Contains('r') || text.Contains('R'))
        {
            // Handle radix numbers
            var parts = text.Split(['r', 'R']);
            var radix = int.Parse(parts[0]);
            var value = Convert.ToInt32(parts[1], radix);
            return new LiteralExpression(value);
        }
        else
        {
            return new LiteralExpression(int.Parse(text));
        }
    }

    /// <summary>
    /// Visit a block context.
    /// </summary>
    public override object? VisitBlock(Smalltalk80Parser.BlockContext context)
    {
        var parameters = new List<string>();

        // Extract block arguments if present
        if (context.blockArgs() != null)
        {
            var argumentNames = context.blockArgs().argumentName();
            if (argumentNames != null)
            {
                foreach (var arg in argumentNames)
                {
                    if (arg != null)
                        parameters.Add(arg.GetText());
                }
            }
        }

        // Extract block temporaries if present
        var temporaries = new List<string>();
        if (context.blockTemps() != null)
        {
            var temporaryNames = context.blockTemps().temporaryName();
            if (temporaryNames != null)
            {
                foreach (var temp in temporaryNames)
                {
                    if (temp != null)
                        temporaries.Add(temp.GetText());
                }
            }
        }

        // Create block body - if no statements, create empty sequence
        SequenceNode body;
        if (context.statements() != null)
        {
            var statements = (List<StatementNode>)Visit(context.statements())!;
            body = new SequenceNode(temporaries, statements);
        }
        else
        {
            body = new SequenceNode(temporaries, new List<StatementNode>());
        }

        return new BlockExpression(parameters, body);
    }

    /// <summary>
    /// Visit a negative number context.
    /// </summary>
    public override object? VisitNegativeNumber(Smalltalk80Parser.NegativeNumberContext context)
    {
        var innerExpression = Visit(context.messageBase());
        if (innerExpression is LiteralExpression literal)
        {
            if (literal.Value is int intValue)
                return new LiteralExpression(-intValue);
            if (literal.Value is double doubleValue)
                return new LiteralExpression(-doubleValue);
        }
        throw new InvalidOperationException("Negative number must be applied to a numeric literal");
    }

    /// <summary>
    /// Visit a string constant context.
    /// </summary>
    public override object? VisitStringConst(Smalltalk80Parser.StringConstContext context)
    {
        var text = context.GetText();
        // Remove quotes and handle escaped quotes
        var content = text.Substring(1, text.Length - 2).Replace("''", "'");
        return new LiteralExpression(content);
    }

    /// <summary>
    /// Visit a character literal context.
    /// </summary>
    public override object? VisitCharacterLiteral(Smalltalk80Parser.CharacterLiteralContext context)
    {
        var text = context.GetText();
        if (text == "$!!")
        {
            return new LiteralExpression('!');
        }
        else
        {
            return new LiteralExpression(text[1]); // Skip the '$'
        }
    }

    /// <summary>
    /// Visit a parenthesized expression context.
    /// </summary>
    public override object? VisitParenthesizedExpr(Smalltalk80Parser.ParenthesizedExprContext context)
    {
        return Visit(context.expression());
    }

    /// <summary>
    /// Visit a statement context.
    /// </summary>
    public override object? VisitStatement(Smalltalk80Parser.StatementContext context)
    {
        if (context.returnStatement() != null)
            return Visit(context.returnStatement());
        if (context.expressionStatement() != null)
            return Visit(context.expressionStatement());

        throw new InvalidOperationException("Unknown statement type");
    }

    /// <summary>
    /// Visit a statements context.
    /// </summary>
    public override object? VisitStatements(Smalltalk80Parser.StatementsContext context)
    {
        var statements = new List<StatementNode>();
        foreach (var stmt in context.statement())
        {
            statements.Add((StatementNode)Visit(stmt)!);
        }
        return statements;
    }

    /// <summary>
    /// Visit a non-keyword expression context (used in keyword arguments).
    /// </summary>
    public override object? VisitNonKeywordExpr(Smalltalk80Parser.NonKeywordExprContext context)
    {
        // Start with the message base
        var current = (ExpressionNode)Visit(context.messageBase())!;

        // Apply unary and binary messages in order
        for (int i = 1; i < context.ChildCount; i++)
        {
            var child = context.GetChild(i);
            if (child is Smalltalk80Parser.UnaryMessageContext unary)
            {
                var selector = unary.unarySelector().GetText();
                current = new UnarySendExpression(current, selector);
            }
            else if (child is Smalltalk80Parser.BinaryMessageContext binary)
            {
                var selector = binary.binarySelector().GetText();
                var argument = (ExpressionNode)Visit(binary.messageBase())!;
                current = new BinarySendExpression(current, selector, argument);
            }
        }

        return current;
    }

    /// <summary>
    /// Visit a symbol constant context.
    /// </summary>
    public override object? VisitSymbolConst(Smalltalk80Parser.SymbolConstContext context)
    {
        if (context.QUOTED_SYMBOL() != null)
        {
            var text = context.QUOTED_SYMBOL().GetText();
            return new LiteralExpression(text[1..]); // Skip the #
        }

        if (context.symbolName() != null)
        {
            var symbolName = context.symbolName().GetText();
            return new LiteralExpression(symbolName);
        }

        throw new InvalidOperationException("Unknown symbol type");
    }

    /// <summary>
    /// Visit an array literal context.
    /// </summary>
    public override object? VisitArrayLiteral(Smalltalk80Parser.ArrayLiteralContext context)
    {
        var elements = new List<object>();

        if (context.arrayElement() != null)
        {
            foreach (var element in context.arrayElement())
            {
                var elementValue = Visit(element);
                if (elementValue is LiteralExpression literal)
                    elements.Add(literal.Value!);
                else
                    elements.Add(elementValue!);
            }
        }

        return new LiteralExpression(elements.ToArray());
    }

    /// <summary>
    /// Visit an array element context.
    /// </summary>
    public override object? VisitArrayElement(Smalltalk80Parser.ArrayElementContext context)
    {
        if (context.literal() != null)
            return Visit(context.literal());

        if (context.symbolName() != null)
        {
            var symbolName = context.symbolName().GetText();
            return new LiteralExpression(symbolName);
        }

        // Handle nested array elements (parenthesized arrays)
        if (context.LEFT_PAREN() != null)
        {
            var elements = new List<object>();
            foreach (var element in context.arrayElement())
            {
                var elementValue = Visit(element);
                if (elementValue is LiteralExpression literal)
                    elements.Add(literal.Value!);
                else
                    elements.Add(elementValue!);
            }
            return new LiteralExpression(elements.ToArray());
        }

        throw new InvalidOperationException("Unknown array element type");
    }

    /// <summary>
    /// Visit a byte array literal context.
    /// </summary>
    public override object? VisitByteArrayLiteral(Smalltalk80Parser.ByteArrayLiteralContext context)
    {
        var numbers = context.NUMBER();
        var bytes = new byte[numbers.Length];
        
        for (int i = 0; i < numbers.Length; i++)
        {
            bytes[i] = byte.Parse(numbers[i].GetText());
        }
        
        return new LiteralExpression(bytes);
    }
}

    /// </summary>
    /// <param name="ctx">The execution context.</param>
    public static bool Value(Context ctx)
    {

        while (ctx.Stack.Count > 0)
        {
            var obj = ctx.Stack.Pop();
            if (obj is BlockContext block)
            {
                var args = popped.ToArray();
                args = args.Length > 0 ? args : Array.Empty<object?>();
                Array.Reverse(args);
                if (args.Length != block.ArgumentCount)
                {
                    ctx.Stack.Push(block);
                    foreach (var item in popped.AsEnumerable().Reverse())
                        ctx.Stack.Push(item);
                    return false;
                }

                var exec = BlockContext.Create(block.HomeContext, block.StartPC, block.ArgumentCount, args, ctx);
                var interpreter = new Interpreter { ExitOnReturn = false };
                var result = interpreter.Evaluate(exec);
                ctx.Stack.Push(result);
                return true;
            }
            popped.Add(obj);
        }

            ctx.Stack.Push(item);
        return false;

        return true;
    }

    /// <summary>
    /// Pushes the number of indexable fields in a <see cref="SmalltalkObject"/>.
    /// </summary>
    /// <param name="ctx">The execution context.</param>
    public static bool Size(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject obj)
            return false;
        ctx.Stack.Push(obj.Fields.Length);
        return true;
    }

    /// <summary>
    /// Pushes the next object after the receiver in the enumeration of all objects.
    /// Used by Object>>nextObject primitive 139.
    /// </summary>
    /// <param name="ctx">The execution context.</param>
    public static bool NextObject(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject obj)
            return false;

        int currentIndex = obj.Oop >> 1;
        var objects = ObjectMemory.Objects;

        // Find the next object in the enumeration
        if (currentIndex + 1 < objects.Count)
        {
            ctx.Stack.Push(objects[currentIndex + 1]);
        }
        else
        {
            // No more objects, return nil
            ctx.Stack.Push(null);
        }

        return true;
    }
}

/// Implements storage management primitives such as instance creation,

    /// Retrieves the pointer field at the given one-based index.
    /// </summary>
    public static bool ObjectAt(Context ctx)
    {
        if (ctx.Stack.Pop() is not int index || ctx.Stack.Pop() is not SmalltalkObject obj)
            return false;
        if (index < 1 || index > obj.Fields.Length)
            return false;
        ctx.Stack.Push(obj.Fields[index - 1]);
        return true;
    }

    /// <summary>
    /// Stores a value into the pointer field at the specified index.
    /// </summary>
    public static bool ObjectAtPut(Context ctx)
    {
        var value = ctx.Stack.Pop();
        if (ctx.Stack.Pop() is not int index || ctx.Stack.Pop() is not SmalltalkObject obj)
            return false;
        if (index < 1 || index > obj.Fields.Length)
            return false;
        obj.Fields[index - 1] = value;
        ctx.Stack.Push(value);
        return true;
    }

    /// <summary>Creates a new instance of a class with no indexable fields.</summary>
    public static bool New(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject cls)
            return false;
        var instance = new SmalltalkObject(cls, 0);
        ctx.Stack.Push(instance);
        return true;
    }

    /// <summary>Creates a new instance of a class with a given field count.</summary>
    public static bool NewWithArg(Context ctx)
    {
        if (ctx.Stack.Pop() is not int size || ctx.Stack.Pop() is not SmalltalkObject cls)
            return false;
        var instance = new SmalltalkObject(cls, size);
        ctx.Stack.Push(instance);
        return true;
    }

    /// <summary>Swaps the identities of two objects.</summary>
    public static bool Become(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject other || ctx.Stack.Pop() is not SmalltalkObject receiver)
            return false;
        ObjectMemory.SwapPointers(receiver, other);
        UpdateContexts(ctx, receiver, other);
        ctx.Stack.Push(receiver);
        return true;
    }

    private static void UpdateContexts(Context? ctx, SmalltalkObject a, SmalltalkObject b)
    {
        for (var c = ctx; c != null; c = c.Sender)
        {
            UpdateArray(c.Temporaries, a, b);
            UpdateArray(c.Arguments, a, b);
            if (ReferenceEquals(c.Receiver, a)) c.Receiver = b;
            else if (ReferenceEquals(c.Receiver, b)) c.Receiver = a;
            var items = c.Stack.ToArray();
            bool changed = false;
            for (int i = 0; i < items.Length; i++)
            {
                if (ReferenceEquals(items[i], a)) { items[i] = b; changed = true; }
                else if (ReferenceEquals(items[i], b)) { items[i] = a; changed = true; }
            }
            if (changed)
            {
                c.Stack.Clear();
                for (int i = items.Length - 1; i >= 0; i--)
                    c.Stack.Push(items[i]);
            }
        }
    }

    private static void UpdateArray(object?[] array, SmalltalkObject a, SmalltalkObject b)
    {
        for (int i = 0; i < array.Length; i++)
        {
            if (ReferenceEquals(array[i], a)) array[i] = b;
            else if (ReferenceEquals(array[i], b)) array[i] = a;
        }
    }

    public static bool InstVarAt(Context ctx) => ObjectAt(ctx);

    public static bool InstVarAtPut(Context ctx) => ObjectAtPut(ctx);

    }
