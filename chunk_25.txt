using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace Smalltalk80.Compiler;

/// <summary>
/// Extracts classes from the Smalltalk-80 system image and generates Phase 3 bootstrap JSON.
/// Phase 3 includes complete method implementations with bytecode, suitable for bootstrap.
/// </summary>
public class Phase3ClassExtractor
{
    /// <summary>
    /// Phase 3 class definition with complete method implementations.
    /// </summary>
    public class Phase3Class
    {
        [JsonPropertyName("phase")]
        public string Phase { get; set; } = "3";
        
        [JsonPropertyName("className")]
        public string ClassName { get; set; } = "";
        
        [JsonPropertyName("superclass")]
        public string Superclass { get; set; } = "Object";
        
        [JsonPropertyName("comment")]
        public string Comment { get; set; } = "";
        
        [JsonPropertyName("instanceVariables")]
        public List<string> InstanceVariables { get; set; } = new();
        
        [JsonPropertyName("classVariables")]
        public List<string> ClassVariables { get; set; } = new();
        
        [JsonPropertyName("poolDictionaries")]
        public List<string> PoolDictionaries { get; set; } = new();
        
        [JsonPropertyName("category")]
        public string Category { get; set; } = "Kernel-Objects";
        
        [JsonPropertyName("methods")]
        public List<Phase3Method> Methods { get; set; } = new();
        
        [JsonPropertyName("classMethods")]
        public List<Phase3Method> ClassMethods { get; set; } = new();
    }
    
    /// <summary>
    /// Phase 3 method definition with bytecode.
    /// </summary>
    public class Phase3Method
    {
        [JsonPropertyName("selector")]
        public string Selector { get; set; } = "";
        
        [JsonPropertyName("source")]
        public string Source { get; set; } = "";
        
        [JsonPropertyName("bytecodes")]
        public List<int> Bytecodes { get; set; } = new();
        
        [JsonPropertyName("literals")]
        public List<object> Literals { get; set; } = new();
        
        [JsonPropertyName("primitive")]
        public int Primitive { get; set; } = 0;
        
        [JsonPropertyName("numArgs")]
        public int NumArgs { get; set; } = 0;
        
        [JsonPropertyName("numTemps")]
        public int NumTemps { get; set; } = 0;
        
        [JsonPropertyName("category")]
        public string Category { get; set; } = "";
    }
    
    /// <summary>
    /// Extracts a class from the system image with all its methods compiled to bytecode.
    /// </summary>
    public static int ExtractPhase3Class(string systemImagePath, string className, string outputPath)
    {
        try
        {
            Console.WriteLine($"Extracting Phase 3 class: {className}");
            
            // Step 1: Extract class structure using SimpleClassExtractor
            var classExtractor = new SimpleClassExtractor(systemImagePath);
            classExtractor.ParseSystemImage();
            
            var classInfo = classExtractor.GetClass(className);
            if (classInfo == null)
            {
                Console.Error.WriteLine($"Class {className} not found in system image");
                return 1;
            }
            
            var phase3Class = new Phase3Class
            {
                ClassName = className,
                Superclass = classInfo.Superclass,
                Comment = classInfo.Comment ?? "",
                InstanceVariables = classInfo.InstanceVariables,
                ClassVariables = classInfo.ClassVariables,
                PoolDictionaries = classInfo.PoolDictionaries,
                Category = classInfo.Category ?? GetCategoryForClass(className)
            };
            
            // Step 2: Extract and compile methods
            var methodExtractor = new SimpleClassExtractor();
            var methods = methodExtractor.ExtractMethodsForClass(systemImagePath, className);
            
            // Step 3: Compile methods to bytecode
            var bytecodeCompiler = new BytecodeCompiler();
            var instanceVars = classInfo.InstanceVariables;
            
            Console.WriteLine($"Compiling {methods.Count} methods for {className}...");
            
            foreach (var method in methods)
            {
                try
                {
                    var phase3Method = new Phase3Method
                    {
                        Selector = method.Selector,
                        Source = method.Source,
                        Category = method.Category ?? ""
                    };
                    
                    // Check if it's a primitive method
                    if (method.IsPrimitive)
                    {
                        phase3Method.Primitive = method.PrimitiveNumber;
                    }
                    
                    // Compile to bytecode
                    var compiledData = bytecodeCompiler.CompileMethod(method.Source, instanceVars);
                    phase3Method.Bytecodes = compiledData.Bytecode.Select(b => (int)b).ToList();
                    
                    // Convert literals to strings (handling Symbol objects)
                    phase3Method.Literals = compiledData.Literals.Select(lit => 
                    {
                        if (lit is Symbol symbol)
                            return symbol.Value;
                        else if (lit is string str)
                            return str;
                        else if (lit is int || lit is double || lit is bool)
                            return lit;
                        else
                            return (object)(lit?.ToString() ?? "nil");
                    }).ToList();
                    
                    phase3Method.NumArgs = compiledData.ArgumentCount;
                    phase3Method.NumTemps = compiledData.TemporaryCount;
                    
                    // Add to appropriate list
                    if (method.IsClassMethod)
                    {
                        phase3Class.ClassMethods.Add(phase3Method);
                        Console.WriteLine($"  ✓ class>>{method.Selector}");
                    }
                    else
                    {
                        phase3Class.Methods.Add(phase3Method);
                        Console.WriteLine($"  ✓ {method.Selector}");
                    }
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"  ✗ {method.Selector}: {ex.Message}");
                    // For methods that fail to compile, add a simple return self bytecode
                    var fallbackMethod = new Phase3Method
                    {
                        Selector = method.Selector,
                        Source = method.Source,
                        Category = method.Category ?? "",
                        Bytecodes = new List<int> { 176 }, // returnSelf
                        Literals = new List<object>(),
                        NumArgs = 0,
                        NumTemps = 0
                    };
                    
                    if (method.IsClassMethod)
                    {
                        phase3Class.ClassMethods.Add(fallbackMethod);
                    }
                    else
                    {
                        phase3Class.Methods.Add(fallbackMethod);
                    }
                }
            }
            
            Console.WriteLine($"Extracted {phase3Class.Methods.Count} instance methods and {phase3Class.ClassMethods.Count} class methods");
            
            // Step 4: Save to JSON file
            var options = new JsonSerializerOptions
            {
                WriteIndented = true,
                DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
            };
            
            var json = JsonSerializer.Serialize(phase3Class, options);
            File.WriteAllText(outputPath, json);
            Console.WriteLine($"Saved Phase 3 class to: {outputPath}");
            
            return 0;
        }
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Error extracting Phase 3 class: {ex.Message}");
            if (ex.StackTrace != null)
            {
                Console.Error.WriteLine(ex.StackTrace);
            }
            return 1;
        }
    }
    
    private static string GetCategoryForClass(string className)
    {
        // Default categories for common classes
        return className switch
        {
            "Object" or "UndefinedObject" or "Boolean" or "True" or "False" => "Kernel-Objects",
            "Magnitude" or "Number" or "Integer" or "SmallInteger" or "Float" or "Fraction" => "Numeric-Numbers",
            "Character" or "String" or "Symbol" => "Collections-Text",
            "Collection" or "Set" or "Dictionary" or "Array" or "OrderedCollection" 
                or "SortedCollection" or "Bag" or "IdentityDictionary" or "MethodDictionary" => "Collections-Abstract",
            "Stream" or "ReadStream" or "WriteStream" or "ReadWriteStream" or "FileStream" => "Collections-Streams",
            "Class" or "Metaclass" or "Behavior" or "ClassDescription" or "CompiledMethod" => "Kernel-Classes",
            "Message" or "MessageNotUnderstood" => "Kernel-Methods",
            "Context" or "MethodContext" or "BlockContext" => "Kernel-Contexts",
            _ => "Kernel-Objects"
        };
    }
}

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;
using Antlr4.Runtime;
using Smalltalk80.Generated;
using Smalltalk80.Parser;

namespace Smalltalk80.Compiler;

/// <summary>
/// Extracts class definitions from the Smalltalk-80 system image and generates Phase 2 bootstrap JSON.
/// This simpler version focuses on class structure without bytecode compilation.
/// </summary>
public class SimpleClassExtractor
{
    private readonly string? systemImagePath;
    private readonly Dictionary<string, ClassInfo> classes = new();
    
    public SimpleClassExtractor(string systemImagePath)
    {
        this.systemImagePath = systemImagePath;
    }
    
    public SimpleClassExtractor()
    {
        this.systemImagePath = null;
    }

    /// <summary>
    /// Gets a specific class by name.
    /// </summary>
    public ClassInfo? GetClass(string className)
    {
        return classes.ContainsKey(className) ? classes[className] : null;
    }
    
    /// <summary>
    /// Parse the system image and extract class information.
    /// </summary>
    public void ParseSystemImage()
    {
        if (systemImagePath == null) 
            throw new InvalidOperationException("System image path not provided");
            
        Console.WriteLine($"Parsing system image: {systemImagePath}");
        
        string input = File.ReadAllText(systemImagePath);
        var inputStream = new AntlrInputStream(input);
        var lexer = new Smalltalk80Lexer(inputStream);
        var tokens = new CommonTokenStream(lexer);
        var parser = new Smalltalk80Parser(tokens);
        
        // Parse the file
        var tree = parser.fileIn();
        
        // Visit the tree to extract classes
        var visitor = new ClassDefBuildingVisitor();
        var fileNode = visitor.Visit(tree) as FileNode;
        
        if (fileNode == null)
        {
            throw new InvalidOperationException("Failed to parse system image");
        }

        ProcessFileNode(fileNode);
        
        Console.WriteLine($"Extracted {classes.Count} classes");
    }

    private void ProcessFileNode(FileNode fileNode)
    {
        foreach (var chunk in fileNode.Chunks)
        {
            if (chunk is ClassDefinitionNode classDef)
            {
                classes[classDef.ClassName] = new ClassInfo
                {
                    Name = classDef.ClassName,
                    Superclass = classDef.Superclass,
                    InstanceVariables = classDef.InstanceVariables.ToList(),
                    ClassVariables = classDef.ClassVariables.ToList(),
                    PoolDictionaries = classDef.PoolDictionaries.ToList(),
                    Category = classDef.Category
                };
            }
        }
    }

    /// <summary>
    /// Generate Phase 2 bootstrap JSON for specified classes.
    /// </summary>
    public string GeneratePhase2Json(params string[] classNames)
    {
        var phase2 = new
        {
            phase = "2-extracted",
            description = $"Classes extracted from system image: {string.Join(", ", classNames)}",
            comments = new[]
            {
                "These classes were extracted from the Smalltalk-80 system image.",
                "This Phase 2 creates the class structure without methods.",
                "Methods should be added in Phase 3."
            },
            classes = new List<object>()
        };

        foreach (var className in classNames)
        {
            if (!classes.ContainsKey(className))
            {
                Console.WriteLine($"Warning: Class {className} not found in system image");
                continue;
            }

            var classInfo = classes[className];
            
            // Check if this class has variable-sized instances
            bool variableSize = IsVariableSizeClass(className, classInfo);
            
            var classEntry = new Dictionary<string, object>
            {
                ["name"] = classInfo.Name,
                ["superclass"] = classInfo.Superclass,
                ["instanceVariables"] = classInfo.InstanceVariables.ToArray()
            };

            // Only add non-empty arrays
            if (classInfo.ClassVariables.Count > 0)
            {
                classEntry["classVariables"] = classInfo.ClassVariables.ToArray();
            }
            
            if (classInfo.PoolDictionaries.Count > 0)
            {
                classEntry["poolDictionaries"] = classInfo.PoolDictionaries.ToArray();
            }
            
            if (variableSize)
            {
                classEntry["variableSize"] = true;
            }
            
            phase2.classes.Add(classEntry);
        }

        var options = new JsonSerializerOptions
        {
            WriteIndented = true,
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            DefaultIgnoreCondition = System.Text.Json.Serialization.JsonIgnoreCondition.WhenWritingNull
        };

        return JsonSerializer.Serialize(phase2, options);
    }

    private bool IsVariableSizeClass(string className, ClassInfo classInfo)
    {
        // Classes that typically have variable-sized instances
        var variableSizeClasses = new HashSet<string>
        {
            "Array", "ByteArray", "String", "Symbol",
            "OrderedCollection", "Set", "Dictionary",
            "MethodDictionary", "IdentityDictionary",
            "ArrayedCollection", "SequenceableCollection",
            "CompiledMethod", "BlockContext", "MethodContext"
        };
        
        // Check if this class or its superclass is known to be variable-sized
        if (variableSizeClasses.Contains(className))
            return true;
            
        // Check superclass chain (simplified - just immediate superclass)
        if (variableSizeClasses.Contains(classInfo.Superclass))
            return true;
            
        return false;
    }

    /// <summary>
    /// Extracts methods for a specific class from the system image.
    /// </summary>
    public List<MethodInfo> ExtractMethodsForClass(string systemImagePath, string className)
    {
        var methods = new List<MethodInfo>();
        
        Console.WriteLine($"Extracting methods for {className} from {systemImagePath}");
        
        // Parse the system image
        var input = File.ReadAllText(systemImagePath);
        var inputStream = new AntlrInputStream(input);
        var lexer = new Smalltalk80Lexer(inputStream);
        var tokens = new CommonTokenStream(lexer);
        var parser = new Smalltalk80Parser(tokens);
        
        var tree = parser.fileIn();
        var visitor = new MethodExtractingVisitor(className);
        visitor.Visit(tree);
        
        return visitor.ExtractedMethods;
    }

    public class ClassInfo
    {
        public string Name { get; set; } = "";
        public string ClassName => Name; // Alias for compatibility
        public string Superclass { get; set; } = "";
        public List<string> InstanceVariables { get; set; } = new();
        public List<string> ClassVariables { get; set; } = new();
        public List<string> PoolDictionaries { get; set; } = new();
        public string? Category { get; set; }
        public string? Comment { get; set; }
    }
    
    /// <summary>
    /// Visitor that extracts methods for a specific class from the system image.
    /// </summary>
    private class MethodExtractingVisitor : Smalltalk80BaseVisitor<object?>
    {
        private readonly string targetClassName;
        private readonly List<MethodInfo> extractedMethods = new();
        private bool inTargetClass = false;
        private bool isClassMethod = false;
        
        public List<MethodInfo> ExtractedMethods => extractedMethods;
        
        public MethodExtractingVisitor(string targetClassName)
        {
            this.targetClassName = targetClassName;
        }
        
        public override object? VisitMethodCategoryChunk(Smalltalk80Parser.MethodCategoryChunkContext context)
        {
            // Check if this is methods for our target class
            var className = context.className()?.GetText();
            var classKeyword = context.IDENTIFIER()?.GetText();
            
            // Determine if we're looking at class methods
            bool isClassSide = classKeyword == "class";
            
            // Handle both instance and class methods
            if (className == targetClassName)
            {
                inTargetClass = true;
                isClassMethod = isClassSide;
                Console.WriteLine($"Found {(isClassMethod ? "class" : "instance")} methods for {targetClassName}");
                
                // Visit all method definitions in this chunk
                var methods = context.methodDef();
                foreach (var method in methods)
                {
                    Visit(method);
                }
                
                inTargetClass = false;
                isClassMethod = false;
            }
            
            return null;
        }
        
        public override object? VisitMethodDef(Smalltalk80Parser.MethodDefContext context)
        {
            if (!inTargetClass) return null;
            
            // Extract method information
            var header = context.methodHeader();
            var selector = ExtractSelector(header);
            var source = context.GetText(); // Get the full method source
            
            // Check for primitive
            var primitiveDecl = context.primitiveDecl();
            bool isPrimitive = primitiveDecl != null;
            int primitiveNumber = 0;
            
            if (isPrimitive && primitiveDecl != null)
            {
                // Extract number from primitive declaration text like "<primitive: 123>"
                var primitiveText = primitiveDecl.GetText();
                var match = System.Text.RegularExpressions.Regex.Match(primitiveText, @"<primitive:\s*(\d+)>");
                if (match.Success)
                {
                    int.TryParse(match.Groups[1].Value, out primitiveNumber);
                }
            }
            
            var methodInfo = new MethodInfo
            {
                Selector = selector,
                Source = source,
                IsPrimitive = isPrimitive,
                PrimitiveNumber = primitiveNumber,
                IsClassMethod = isClassMethod
            };
            
            extractedMethods.Add(methodInfo);
            Console.WriteLine($"  Extracted {(isClassMethod ? "class" : "instance")} method: {selector}");
            
            return null;
        }
        
        private string ExtractSelector(Smalltalk80Parser.MethodHeaderContext header)
        {
            if (header.unarySelector() != null)
            {
                return header.unarySelector().GetText();
            }
            else if (header.binarySelector() != null)
            {
                return header.binarySelector().GetText();
            }
            else
            {
                // Keyword selector - concatenate keyword parts without spaces
                var keywords = header.identifierColon();
                return string.Join("", keywords.Select(k => k.GetText()));
            }
        }
    }
}

        }
    }
}

// file: Smalltalk80.ExtractClass/ClassExtractor.cs
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;
using System.Text.Json.Serialization;
using Smalltalk80.Compiler;

namespace Smalltalk80.ExtractClass
{
    /// <summary>
    /// Represents an extracted Smalltalk class with full bytecode compilation.
    /// </summary>
    public class ExtractedClass
    {
        public string ClassName { get; set; } = "";
        public string Superclass { get; set; } = "Object";
        public List<string> InstanceVariables { get; set; } = new();
        public List<string> ClassVariables { get; set; } = new();
        public List<string> PoolDictionaries { get; set; } = new();
        public string? Category { get; set; }
        
        public List<ExtractedMethod> InstanceMethods { get; set; } = new();
        public List<ExtractedMethod> ClassMethods { get; set; } = new();
        
        public ExtractionStatistics Statistics { get; set; } = new();
        
        [JsonIgnore]
        public int InstanceMethodCount => InstanceMethods.Count;
        
        [JsonIgnore]
        public int ClassMethodCount => ClassMethods.Count;
    }
    
    /// <summary>
    /// Represents an extracted method with compiled bytecode.
    /// </summary>
    public class ExtractedMethod
    {
        public string Selector { get; set; } = "";
        public string Source { get; set; } = "";
        public string? Category { get; set; }
        
        public int ArgumentCount { get; set; }
        public int TemporaryCount { get; set; }
        public bool IsPrimitive { get; set; }
        public int PrimitiveNumber { get; set; }
        
        public string Bytecode { get; set; } = "";
        public object[] Literals { get; set; } = Array.Empty<object>();
        
        public string? CompilationError { get; set; }
        
        [JsonIgnore]
        public bool CompiledSuccessfully => CompilationError == null;
    }
    
    /// <summary>
    /// Statistics about the extraction process.
    /// </summary>
    public class ExtractionStatistics
    {
        public int TotalMethods { get; set; }
        public int SuccessfullyCompiled { get; set; }
        public int FailedCompilation { get; set; }
        public double CompilationRate { get; set; }
        public DateTime ExtractionTime { get; set; } = DateTime.UtcNow;
    }
    
    /// <summary>
    /// Extracts complete class definitions from Smalltalk source with compiled bytecode.
    /// </summary>
    public class ClassExtractor
    {
        private readonly BytecodeCompiler bytecodeCompiler;
        
        public ClassExtractor()
        {
            bytecodeCompiler = new BytecodeCompiler();
        }
        
        /// <summary>
        /// Extract a class from the system image with full bytecode compilation.
        /// </summary>
        public ExtractedClass? ExtractClass(string systemImagePath, string className)
        {
            Console.WriteLine($"\nExtracting class: {className}");
            
            try
            {
                // Step 1: Parse system image to get class structure
                Console.WriteLine("  Parsing system image...");
                var extractor = new SimpleClassExtractor(systemImagePath);
                extractor.ParseSystemImage();
                
                var classInfo = extractor.GetClass(className);
                if (classInfo == null)
                {
                    Console.WriteLine($"  Error: Class {className} not found in system image");
                    return null;
                }
                
                Console.WriteLine($"  Found class structure:");
                Console.WriteLine($"    Superclass: {classInfo.Superclass}");
                Console.WriteLine($"    Instance vars: {string.Join(", ", classInfo.InstanceVariables)}");
                Console.WriteLine($"    Class vars: {string.Join(", ", classInfo.ClassVariables)}");
                
                // Step 2: Extract methods
                Console.WriteLine("  Extracting methods...");
                var methods = extractor.ExtractMethodsForClass(systemImagePath, className);
                Console.WriteLine($"  Found {methods.Count} methods");
                
                // Create extracted class data
                var extracted = new ExtractedClass
                {
                    ClassName = className,
                    Superclass = classInfo.Superclass,
                    InstanceVariables = classInfo.InstanceVariables,
                    ClassVariables = classInfo.ClassVariables,
                    PoolDictionaries = classInfo.PoolDictionaries,
                    Category = classInfo.Category,
                    InstanceMethods = new List<ExtractedMethod>(),
                    ClassMethods = new List<ExtractedMethod>(),
                    Statistics = new ExtractionStatistics()
                };
                
                // Step 3: Compile each method
                Console.WriteLine($"  Compiling {methods.Count} methods...");
                var instanceVars = classInfo.InstanceVariables;
                int successCount = 0;
                int failureCount = 0;
                
                foreach (var method in methods)
                {
                    try
                    {
                        // Compile the method
                        var compiledData = bytecodeCompiler.CompileMethod(method.Source, instanceVars);
                        
                        var extractedMethod = new ExtractedMethod
                        {
                            Selector = method.Selector,
                            Source = method.Source,
                            Category = method.Category,
                            ArgumentCount = compiledData.ArgumentCount,
                            TemporaryCount = compiledData.TemporaryCount,
                            Bytecode = BitConverter.ToString(compiledData.Bytecode).Replace("-", " "),
                            Literals = compiledData.Literals,
                            IsPrimitive = compiledData.IsPrimitive,
                            PrimitiveNumber = compiledData.PrimitiveNumber
                        };
                        
                        // Add to appropriate collection
                        if (method.IsClassMethod)
                        {
                            extracted.ClassMethods.Add(extractedMethod);
                        }
                        else
                        {
                            extracted.InstanceMethods.Add(extractedMethod);
                        }
                        
                        successCount++;
                        Console.Write(".");
                    }
                    catch (Exception ex)
                    {
                        // Add method with error information
                        var failedMethod = new ExtractedMethod
                        {
                            Selector = method.Selector,
                            Source = method.Source,
                            Category = method.Category,
                            CompilationError = ex.Message
                        };
                        
                        if (method.IsClassMethod)
                        {
                            extracted.ClassMethods.Add(failedMethod);
                        }
                        else
                        {
                            extracted.InstanceMethods.Add(failedMethod);
                        }
                        
                        failureCount++;
                        Console.Write("x");
                    }
                }
                
                Console.WriteLine(); // New line after progress dots
                
                // Update statistics
                extracted.Statistics.TotalMethods = methods.Count;
                extracted.Statistics.SuccessfullyCompiled = successCount;
                extracted.Statistics.FailedCompilation = failureCount;
                extracted.Statistics.CompilationRate = methods.Count > 0 ? 
                    (double)successCount / methods.Count * 100 : 0;
                
                // Sort methods by category and selector for better organization
                extracted.InstanceMethods = extracted.InstanceMethods
                    .OrderBy(m => m.Category ?? "")
                    .ThenBy(m => m.Selector)
                    .ToList();
                    
                extracted.ClassMethods = extracted.ClassMethods
                    .OrderBy(m => m.Category ?? "")
                    .ThenBy(m => m.Selector)
                    .ToList();
                
                // Report results
                Console.WriteLine($"  Instance methods: {extracted.InstanceMethods.Count}");
                Console.WriteLine($"  Class methods: {extracted.ClassMethods.Count}");
                Console.WriteLine($"  Compilation success rate: {extracted.Statistics.CompilationRate:F1}%");
                
                if (failureCount > 0)
                {
                    Console.WriteLine($"  Warning: {failureCount} methods failed to compile");
                }

        foreach (var kvp in globals)

        // Check if there's a 'thisClass' field pointing to the instance side
        if (classObj.TryGetField("thisClass", out var thisClass) && thisClass is SmalltalkObject thisClassObj)

    /// <summary>
    /// Provides a string description of an object.
    /// </summary>
    private static string DescribeObject(VirtualMachine vm, SmalltalkObject? obj)
    {
        if (obj == null)
            return "nil";
            
        // Check if it's a known special object
        // TODO: Check against global objects once ObjectMemory is updated
            
        // Check if it's a class
        var className = GetClassName(vm, obj);
        if (!string.IsNullOrEmpty(className))
            return $"<Class: {className}>";
            
        // Check if it's a small integer
        if (obj.IsSmallInteger)
            return obj.SmallIntegerValue.ToString();
            
        // Check if it's a string
        if (obj.TryGetStringValue(out var strValue))
            return $"'{strValue}'";
            
        // Check if it's a symbol
        // Symbol would need special handling
            
        // Default description
        return $"<{obj.GetType().Name}@{obj.GetHashCode():X}>";
    }

    /// <summary>
    /// Find the Smalltalk SystemDictionary by looking for it in globals.

        // We need to bootstrap - find "Smalltalk" using the VM primitive
        var smalltalkOop = vm.PrimitiveLookupGlobal("Smalltalk");
        if (smalltalkOop > 0)
        {
            return ObjectMemory.FromOop(smalltalkOop);
        }
        
        // Fallback: look for any SystemDictionary instance
        foreach (var obj in ObjectMemory.Objects)
        {
            if (obj != null && IsSystemDictionary(obj))

    private bool IsSystemDictionary(SmalltalkObject obj)

        // Get the class name by checking if it's registered as SystemDictionary
        var systemDictClassOop = vm.PrimitiveLookupGlobal("SystemDictionary");
        if (systemDictClassOop > 0 && obj.ClassPointer.Oop == systemDictClassOop)
        {
            return true;
        }
        
        return false;
    }
    
    /// <summary>
    /// Find classes by examining the Smalltalk dictionary's contents.
    /// Dictionary structure: parallel arrays or alternating key-value pairs.
    /// </summary>
    private List<(SmalltalkObject obj, string name)> FindClassesFromSmalltalkDict()
    {
        var classes = new List<(SmalltalkObject, string)>();
        
        if (smalltalkDict == null) return classes;
        
        // Dictionary internal structure varies, but typically has:
        // - A tally field (number of entries)
        // - Either parallel arrays (keys, values) or alternating entries
        
        // Try to interpret as a dictionary with parallel arrays first
        if (TryGetDictionaryContents(smalltalkDict, out var entries))
        {
            foreach (var (key, value) in entries)
            {
                if (IsClassObject(value))

                    classes.Add((value, key));
                }
            }

    /// <summary>
    /// Try to extract key-value pairs from a dictionary object.
    /// </summary>
    private bool TryGetDictionaryContents(SmalltalkObject dict, out List<(string key, SmalltalkObject value)> entries)
    {
        entries = new List<(string, SmalltalkObject)>();
        
        // Dictionaries typically have a structure like:
        // Field 0: tally (number of entries)
        // Field 1+: either alternating key-value or parallel arrays
        
        if (dict.Fields.Length < 2) return false;
        
        // Our dictionary implementation uses alternating key-value pairs after tally field
        // Dictionary structure: [tally, key1, value1, key2, value2, ...]
        if (dict.Fields.Length > 1)
        {
            // Start from field 1 (skip tally at field 0)
            for (int i = 1; i < dict.Fields.Length - 1; i += 2)
            {
                var keyObj = dict.Fields[i];
                var valueObj = dict.Fields[i + 1];
                
                // Skip null entries (empty slots)
                if (keyObj == null || valueObj == null)
                    continue;
                
                if (keyObj is SmalltalkObject keySymbol && valueObj is SmalltalkObject value)
                {
                    var keyStr = ExtractSymbolString(keySymbol);
                    if (!string.IsNullOrEmpty(keyStr))
                    {
                        entries.Add((keyStr, value));
                    }
                }
            }
        }
        
        return entries.Count > 0;
    }

    /// <summary>
    /// Determine if an object is a class by examining its structure.
    /// Classes have: superclass (slot 0), methodDict (slot 1), format (slot 2)
    /// </summary>
    private bool IsClassObject(SmalltalkObject obj)
    {
        if (obj == null || obj.Fields.Length < 3)
            return false;
            
        // Check if the object's class is a metaclass or Class
        if (obj.ClassPointer != null)
        {
            var classOop = vm.PrimitiveLookupGlobal("Class");
            var metaclassOop = vm.PrimitiveLookupGlobal("Metaclass");
            
            // If this object's class's class is Metaclass, then this object is a class
            // (because its class is a metaclass)
            if (obj.ClassPointer.ClassPointer != null)
            {
                if (obj.ClassPointer.ClassPointer.Oop == metaclassOop)
                    return true;
            }
            
            // If this object is a direct instance of Class, it's a class
            if (obj.ClassPointer.Oop == classOop)
                return true;
                
            // Check if the class pointer points to a metaclass
            // Note: Without the ability to enumerate globals, we can't check for metaclass naming pattern
            // This is a limitation of the new API but acceptable for the pretty printer
        }
        
        // Fallback: Check if slot 1 looks like a method dictionary
        if (obj.Fields.Length > 1)
        {
            var methodDictField = obj.Fields[1];
            if (methodDictField is SmalltalkObject methodDict)
            {
                // Method dictionaries should have structure
                return methodDict.Fields.Length >= 2;
            }
        }
        
        return false;
    }

    /// </summary>

        // Structure: [tally, key1, value1, key2, value2, ...]
        // Keys are selectors (Symbols) at odd indices, values are CompiledMethods at even indices
        int count = 0;
        
        // Get nil for comparison
        var nilOop = vm.PrimitiveNil();
        
        // Values are at even indices starting from 2: [tally@0, key@1, value@2, key@3, value@4, ...]
        for (int i = 2; i < methodDict.Fields.Length; i += 2)
        {
            var value = methodDict.Fields[i];
            
            // Skip if nil or null
            if (value == null)
                continue;
                
            if (value is SmalltalkObject valueObj)
            {
                // Skip if it's nil
                if (valueObj.Oop == nilOop)
                    continue;
                    
                // Check if it's a CompiledMethod
                if (IsCompiledMethod(valueObj))

                    count++;
                }
            }
        }
        
        return count;
