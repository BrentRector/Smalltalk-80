    {
        var context = interpreter.CurrentContext;
        // Blue Book canonical encoding:
        // Bits 6-7: Argument count (00=0, 01=1, 10=2, 11=N args)
        // Bits 2-5: Literal frame index (0-15)
        // Bits 0-1: Unused/reserved
        
        var argCount = (bytecode >> 6) & 0x3; // Extract bits 6-7
        var selectorIndex = (bytecode >> 2) & 0xF; // Extract bits 2-5
        
        if (selectorIndex >= context.Method.Literals.Length)
        {
            throw new InvalidOperationException($"Selector index {selectorIndex} out of bounds for literal frame (size: {context.Method.Literals.Length})");
        }
        
        if (!(context.Method.Literals[selectorIndex] is Symbol selector))
        {
            throw new InvalidOperationException($"Literal at index {selectorIndex} is not a Symbol: {context.Method.Literals[selectorIndex]?.GetType()}");
        }
        
        // Handle special case for 11xx_xxxx (bits 6-7 = 11) - variable argument count
        if (argCount == 3) // 11 binary
        {
            // For now, treat as 0 arguments - proper implementation would need more context
            // TODO: Implement variable argument count handling
            argCount = 0;
        }
        
        interpreter.PerformSend(selector, argCount, false);

// file: Smalltalk80/ByteCodes/BytecodeTable.cs
using System;
using System.Collections.Generic;

namespace Smalltalk80.ByteCodes;

/// <summary>
///     Provides access to executable implementations for all 256 Smalltalk-80
///     bytecodes defined by the interpreter documentation.
/// </summary>
public static class BytecodeTable
{
    /// <summary>
    ///     Map from bytecode value to the operation that implements it.
    /// </summary>
    public static IReadOnlyList<IBytecodeOperation> Operations { get; } = CreateTable();

    private static IReadOnlyList<IBytecodeOperation> CreateTable()
    {
        var ops = new IBytecodeOperation[256];

        // 0-15: push receiver variable
        for (byte i = 0; i < 16; i++)
        {
            ops[i] = new Stack.PushReceiverVariableOperation(i);
        }

        // 16-31: push temporary location
        for (byte i = 0; i < 16; i++)
        {
            ops[16 + i] = new Stack.PushTemporaryVariableOperation(i);
        }

        // 32-63: push literal constant
        for (byte i = 0; i < 32; i++)
        {
            ops[32 + i] = new Stack.PushLiteralConstantOperation(i);
        }

        // 64-95: push literal variable
        for (byte i = 0; i < 32; i++)
        {
            ops[64 + i] = new Stack.PushLiteralVariableOperation(i);
        }

        // 96-103: store and pop receiver variable
        for (byte i = 0; i < 8; i++)
        {
            ops[96 + i] = new Stack.StoreAndPopReceiverVariableOperation(i);
        }

        // 104-111: store and pop temporary variable
        for (byte i = 0; i < 8; i++)
        {
            ops[104 + i] = new Stack.StoreAndPopTemporaryVariableOperation(i);
        }

        // 112-119: push receiver and special constants (canonical Blue Book)
        ops[112] = new Stack.PushSelfOperation();
        // 113-119 are NOT push true/false/nil in canonical Blue Book!
        // Those must be accessed via literal frame (32-63)
        // These are actually undefined in canonical Blue Book
        ops[113] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[114] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[115] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[116] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[117] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[118] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[119] = new NoOpOperation(); // Not defined in canonical Blue Book

        // 120-123: return operations (canonical Blue Book)
        ops[120] = new Return.ReturnConstantOperation(Return.ReturnConstantOperation.Kind.Self);
        // 121-123 are NOT returnTrue/False/Nil in canonical Blue Book!
        ops[121] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[122] = new NoOpOperation(); // Not defined in canonical Blue Book
        ops[123] = new NoOpOperation(); // Not defined in canonical Blue Book

        // 124-125: return stack top from message or block
        ops[124] = new Return.ReturnTopOperation(isBlock: false);
        ops[125] = new Return.ReturnTopOperation(isBlock: true);

        // 126-127 unused
        ops[126] = new NoOpOperation();
        ops[127] = new NoOpOperation();

        // 128: extended push
        ops[128] = new Stack.ExtendedPushOperation();

        // 129: extended store
        ops[129] = new Stack.ExtendedStoreOperation(popAfter: false);

        // 130: extended store and pop
        ops[130] = new Stack.ExtendedStoreOperation(popAfter: true);

        // 131: single extended send
        ops[131] = new Send.SingleExtendedSendOperation(superSend: false);

        // 132: double extended send
        ops[132] = new Send.DoubleExtendedSendOperation(superSend: false);

        // 133: single extended super
        ops[133] = new Send.SingleExtendedSendOperation(superSend: true);

        // 134: double extended super
        ops[134] = new Send.DoubleExtendedSendOperation(superSend: true);

        // 135: pop stack
        ops[135] = new Stack.PopStackOperation();

        // 136: duplicate stack top
        ops[136] = new Stack.DuplicateTopOperation();

        // 137: push active context
        ops[137] = new Stack.PushActiveContextOperation();

        // 138-143: unused
        for (byte i = 138; i <= 143; i++)
        {
            ops[i] = new NoOpOperation();
        }

        // 144-151: short jumps
        for (byte i = 0; i < 8; i++)
        {
            ops[144 + i] = new Jump.ShortJumpOperation((byte)(i + 1));
        }

        // 152-159: pop and jump on false short
        for (byte i = 0; i < 8; i++)
        {
            ops[152 + i] = new Jump.PopJumpFalseShortOperation((byte)(i + 1));
        }

        // 160-167: long jump
        for (byte i = 0; i < 8; i++)
        {
            ops[160 + i] = new Jump.LongJumpOperation(i);
        }

        // 168-171: pop and jump on true long
        for (byte i = 0; i < 4; i++)
        {
            ops[168 + i] = new Jump.PopJumpLongOperation(true, i);
        }

        // 172-175: pop and jump on false long
        for (byte i = 0; i < 4; i++)
        {
            ops[172 + i] = new Jump.PopJumpLongOperation(false, i);
        }

        // 176-191: send literal selector with 0 args (Blue Book)
        for (byte i = 0; i < 16; i++)
        {
            ops[176 + i] = new Send.LiteralSelectorSendOperation(i, 0);
        }

        // 192-207: send literal selector with 1 arg (Blue Book)
        for (byte i = 0; i < 16; i++)
        {
            ops[192 + i] = new Send.LiteralSelectorSendOperation(i, 1);
        }

        // 208-223: send literal selector with 2 args (Blue Book)
        for (byte i = 0; i < 16; i++)
        {
            ops[208 + i] = new Send.LiteralSelectorSendOperation(i, 2);
        }

        // 224-239: send arithmetic selector (+, -, <, >, etc.)
        for (byte i = 0; i < 16; i++)
        {
            ops[224 + i] = new Send.ArithmeticSelectorSendOperation(i);
        }

        // 240-255: send special selector (at:, at:put:, size, etc.)
        for (byte i = 0; i < 16; i++)
        {
            ops[240 + i] = new Send.SpecialSelectorSendOperation(i);
        }

        return ops;
    }

    private sealed class NoOpOperation : IBytecodeOperation
    {
        public void Execute(Interpreter interpreter, byte opcode) { }
    }

/// <summary>
///     Represents an execution context for a Smalltalk method.
/// </summary>
public class Context
{
    /// <summary>
    ///     Gets or sets the sending context. This will be <c>null</c> for the
    ///     top-level context.
    /// </summary>
    public Context? Sender { get; init; }

    /// <summary>
    ///     Gets the compiled method being executed.
    /// </summary>
    public required CompiledMethod Method { get; init; }

    /// <summary>
    ///     Gets the receiver object for the method.
    /// </summary>
    public SmalltalkObject Receiver { get; set; } = null!;

    /// <summary>
    ///     Gets the array of argument values passed to the method.
    /// </summary>
    public required object?[] Arguments { get; init; } = Array.Empty<object?>();

    /// <summary>
    ///     Gets the storage for temporary variables defined by the method.
    /// </summary>
    public required object?[] Temporaries { get; init; }

    /// <summary>
    ///     Gets the evaluation stack used while executing the method.
    /// </summary>
    public required Stack<object?> Stack { get; init; }

    /// <summary>
    ///     Gets or sets the program counter within the method.
    /// </summary>
    public int PC { get; set; }

    /// <summary>
    ///     Creates a new execution context for sending a message.
    /// </summary>
    /// <param name="selector">The selector of the method to invoke.</param>
    /// <param name="argCount">The number of arguments being sent.</param>
    /// <param name="sender">The calling context.</param>
    /// <returns>A newly initialized <see cref="Context"/>.</returns>
    public static Context CreateFor(Symbol selector, int argCount, Context sender)
    {
        var cls = sender.Receiver.ClassPointer ??
                  throw new InvalidOperationException("Receiver has no class pointer");
        var method = cls.Lookup(selector) ??
                     throw new InvalidOperationException($"Method {selector} not found");

        var args = new object?[argCount];
        for (int i = argCount - 1; i >= 0; i--) {
            args[i] = sender.Stack.Pop();
        }

        return new MethodContext {
            Sender = sender,
            Method = method,
            Receiver = sender.Receiver,
            Arguments = args,
            Temporaries = new object?[method.TemporariesCount],
            Stack = new Stack<object?>(),
            PC = 0
        };
    }
}
