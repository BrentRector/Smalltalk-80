namespace Smalltalk80.Core;

/// <summary>
/// Provides bootstrap data for initializing the Smalltalk VM.
/// Implementations can load data from JSON files, embedded resources, or other sources.
/// </summary>
public interface IBootstrapProvider
{
    /// <summary>
    /// Loads the Phase 1 bootstrap data (VM primitives for core class creation).
    /// </summary>
    /// <returns>The Phase 1 bootstrap configuration.</returns>
    Task<Phase1BootstrapData> LoadPhase1Async();

    /// <summary>
    /// Loads the Phase 2 bootstrap data (messaging-based class and method creation).
    /// </summary>
    /// <returns>The Phase 2 bootstrap configuration.</returns>
    Task<Phase2BootstrapData> LoadPhase2Async();

    /// <summary>
    /// Loads the Phase 3 bootstrap data (file loading for complete system).
    /// </summary>
    /// <returns>The Phase 3 bootstrap configuration.</returns>
    Task<Phase3BootstrapData> LoadPhase3Async();

    /// <summary>
    /// Validates that the bootstrap data is consistent and complete.
    /// </summary>
    /// <returns>Validation results with any errors or warnings.</returns>
    Task<BootstrapValidationResult> ValidateAsync();
}

/// <summary>
/// Contains Phase 1 bootstrap operations using VM primitives.
/// </summary>
public record Phase1BootstrapData
{
    /// <summary>
    /// Gets the list of VM primitive operations to execute.
    /// </summary>
    public IReadOnlyList<PrimitiveOperation> Operations { get; init; } = Array.Empty<PrimitiveOperation>();

    /// <summary>
    /// Gets the validation checkpoints to verify after bootstrap.
    /// </summary>
    public IReadOnlyList<ValidationCheckpoint> Checkpoints { get; init; } = Array.Empty<ValidationCheckpoint>();
}

/// <summary>
/// Represents a single VM primitive operation in Phase 1 bootstrap.
/// </summary>

{

    /// Gets the operation type (e.g., "newClass", "setSlot", "installMethod").

    public string Operation { get; init; } = string.Empty;

    /// <summary>

    /// </summary>
    public string? Target { get; init; }

    /// <summary>
    /// Gets the result variable name for operations that create new objects.
    /// </summary>
    public string? Result { get; init; }

    /// <summary>
    /// Gets operation-specific parameters.
    /// </summary>
    public IReadOnlyDictionary<string, object> Parameters { get; init; } = new Dictionary<string, object>();

    /// <summary>
    /// Gets an optional comment describing this operation.
    /// </summary>
    public string? Comment { get; init; }
}

namespace Smalltalk80.Core;

/// <summary>
/// Manages the allocation, storage, and garbage collection of Smalltalk objects in memory.
/// This interface abstracts the object memory management from the VM implementation.
/// </summary>
public interface IObjectMemory
{
    /// <summary>
    /// Allocates a new object with the specified class and field count.
    /// </summary>
    /// <param name="classObject">The class of the object to create.</param>
    /// <param name="fieldCount">The number of instance variable fields.</param>
    /// <returns>The newly allocated object.</returns>
    ISmalltalkObject AllocateObject(ISmalltalkClass classObject, int fieldCount);

    /// <summary>
    /// Allocates a new indexed object (like Array) with the specified size.
    /// </summary>
    /// <param name="classObject">The class of the indexed object.</param>
    /// <param name="size">The number of indexed slots.</param>
    /// <returns>The newly allocated indexed object.</returns>
    ISmalltalkObject AllocateIndexedObject(ISmalltalkClass classObject, int size);

    /// <summary>
    /// Allocates a new byte object (like String or Symbol) with the specified size.
    /// </summary>
    /// <param name="classObject">The class of the byte object.</param>
    /// <param name="size">The number of bytes.</param>
    /// <returns>The newly allocated byte object.</returns>
    ISmalltalkObject AllocateByteObject(ISmalltalkClass classObject, int size);

    /// <summary>
    /// Gets an object by its object identifier (OOP - Object-Oriented Pointer).
    /// </summary>
    /// <param name="oop">The object identifier.</param>
    /// <returns>The object, or null if not found.</returns>
    ISmalltalkObject? GetObject(int oop);

    /// <summary>
    /// Registers an object and returns its OOP.
    /// </summary>
    /// <param name="obj">The object to register.</param>
    /// <returns>The object's OOP.</returns>
    int RegisterObject(ISmalltalkObject obj);

    /// <summary>
    /// Performs garbage collection to reclaim unused memory.
    /// </summary>
    /// <returns>The number of objects collected.</returns>
    int CollectGarbage();

    /// <summary>
    /// Gets the current memory statistics.
    /// </summary>
    MemoryStatistics GetStatistics();

    /// <summary>
    /// Creates a memory snapshot for debugging or persistence.
    /// </summary>
    /// <returns>A snapshot of the current object memory state.</returns>
    IMemorySnapshot CreateSnapshot();

    /// <summary>
    /// Restores object memory from a snapshot.
    /// </summary>
    /// <param name="snapshot">The snapshot to restore from.</param>
    void RestoreSnapshot(IMemorySnapshot snapshot);
}

/// <summary>
/// Represents memory usage statistics.
/// </summary>
public record MemoryStatistics
{
    /// <summary>
    /// Gets the total number of allocated objects.
    /// </summary>
    public int TotalObjects { get; init; }

    /// <summary>
    /// Gets the total memory used in bytes.
    /// </summary>
    public long UsedMemory { get; init; }

    /// <summary>
    /// Gets the total available memory in bytes.
    /// </summary>
    public long AvailableMemory { get; init; }

    /// <summary>
    /// Gets the number of garbage collections performed.
    /// </summary>
    public int GarbageCollectionCount { get; init; }

    /// <summary>
    /// Gets the breakdown of objects by class.
    /// </summary>
    public IReadOnlyDictionary<string, int> ObjectsByClass { get; init; } = new Dictionary<string, int>();
}

/// <summary>
/// Represents a snapshot of object memory state.
/// </summary>
public interface IMemorySnapshot
{
    /// <summary>
    /// Gets the timestamp when the snapshot was created.
    /// </summary>
    DateTime Timestamp { get; }

    /// <summary>
    /// Gets the serialized object data.
    /// </summary>
    byte[] Data { get; }

    /// <summary>
    /// Gets metadata about the snapshot.
    /// </summary>
    IReadOnlyDictionary<string, object> Metadata { get; }
}

// file: Core/Smalltalk80.Core/ISmalltalkObject.cs
namespace Smalltalk80.Core;

/// <summary>
/// Represents the fundamental contract for all Smalltalk objects in the VM.
/// Every object has a class pointer and can contain fields (instance variables).
/// </summary>
public interface ISmalltalkObject
{
    /// <summary>
    /// Gets or sets the class of this object.
    /// Every Smalltalk object has a class, including classes themselves (whose class is a metaclass).
    /// </summary>
    ISmalltalkClass Class { get; set; }

    /// <summary>
    /// Gets the object identifier (OOP - Object-Oriented Pointer) for this object.
    /// </summary>
    int ObjectId { get; }

    /// <summary>
    /// Gets the number of named instance variable fields in this object.
    /// </summary>
    int FieldCount { get; }

    /// <summary>
    /// Gets or sets a named instance variable field by index.
    /// </summary>
    /// <param name="index">The zero-based field index.</param>
    /// <returns>The field value.</returns>
    ISmalltalkObject? this[int index] { get; set; }

    /// <summary>
    /// Gets or sets a named instance variable field by name.
    /// </summary>
    /// <param name="fieldName">The name of the instance variable.</param>
    /// <returns>The field value.</returns>
    ISmalltalkObject? this[string fieldName] { get; set; }

    /// <summary>
    /// Determines if this object is an indexed object (like Array).
    /// </summary>
    bool IsIndexed { get; }

    /// <summary>
    /// Gets the size of indexed fields for indexed objects.
    /// </summary>
    int IndexedSize { get; }

    /// <summary>
    /// Gets or sets an indexed field (for Array-like objects).
    /// </summary>
    /// <param name="index">The zero-based index.</param>
    /// <returns>The indexed field value.</returns>
    ISmalltalkObject? GetIndexed(int index);
    
    /// <summary>
    /// Sets an indexed field value.
    /// </summary>
    void SetIndexed(int index, ISmalltalkObject? value);

    /// <summary>
    /// Determines if this object is a byte object (like String or Symbol).
    /// </summary>
    bool IsByteObject { get; }

    /// <summary>
    /// Gets the byte data for byte objects.
    /// </summary>
    byte[]? GetBytes();

    /// <summary>
    /// Sets the byte data for byte objects.
    /// </summary>
    void SetBytes(byte[] bytes);

    /// <summary>
    /// Creates a shallow copy of this object.
    /// </summary>
    ISmalltalkObject ShallowCopy();

    /// <summary>
    /// Gets a string representation for debugging.
    /// </summary>
    string ToDebugString();
}

/// <summary>
/// Represents a Smalltalk class object.
/// Classes are special objects that can create instances and contain method dictionaries.
/// </summary>
public interface ISmalltalkClass : ISmalltalkObject
{
    /// <summary>
    /// Gets or sets the superclass of this class.
    /// ProtoObject is the only class with a null superclass.
    /// </summary>
    ISmalltalkClass? Superclass { get; set; }

    /// <summary>
    /// Gets or sets the method dictionary containing this class's methods.
    /// </summary>
    IMethodDictionary MethodDictionary { get; set; }

    /// <summary>
    /// Gets or sets the class name symbol.
    /// </summary>
    ISymbol Name { get; set; }

    /// <summary>
    /// Gets or sets the array of instance variable names.
    /// </summary>
    ISymbol[] InstanceVariables { get; set; }

    /// <summary>
    /// Gets or sets the class variable dictionary (at field index 6).
    /// </summary>
    IDictionary<ISymbol, ISmalltalkObject>? ClassVariables { get; set; }

    /// <summary>
    /// Gets or sets the shared pool dictionaries.
    /// </summary>
    IDictionary<ISymbol, ISmalltalkObject>? PoolDictionaries { get; set; }

    /// <summary>
    /// Gets or sets the format of instances of this class.
    /// Format encodes: indexed/non-indexed, byte/pointer, field count.
    /// </summary>
    int Format { get; set; }

    /// <summary>
    /// Gets the metaclass of this class (for class-side methods).
    /// </summary>
    ISmalltalkClass Metaclass { get; }

    /// <summary>
    /// Looks up a method by selector in this class or its superclasses.
    /// </summary>
    /// <param name="selector">The method selector to find.</param>
    /// <returns>The compiled method, or null if not found.</returns>
    ICompiledMethod? LookupMethod(ISymbol selector);

    /// <summary>
    /// Adds a method to this class's method dictionary.
    /// </summary>
    /// <param name="selector">The method selector.</param>
    /// <param name="method">The compiled method.</param>
    void AddMethod(ISymbol selector, ICompiledMethod method);

    /// <summary>
    /// Creates a new instance of this class.
    /// </summary>
    /// <returns>A new instance with fields initialized to nil.</returns>
    ISmalltalkObject NewInstance();

    /// <summary>
    /// Creates a new indexed instance of this class.
    /// </summary>
    /// <param name="size">The number of indexed slots.</param>
    /// <returns>A new indexed instance.</returns>
    ISmalltalkObject NewInstance(int size);

    /// <summary>
    /// Determines if this class is a subclass of another class.
    /// </summary>
    /// <param name="other">The potential superclass.</param>
    /// <returns>True if this class inherits from the other class.</returns>
    bool IsSubclassOf(ISmalltalkClass other);
}

/// <summary>
/// Represents a compiled Smalltalk method.

public interface ICompiledMethod : ISmalltalkObject

    /// <summary>

    /// <summary>
    /// Gets the literal objects referenced by this method.
    /// </summary>
    ISmalltalkObject[] Literals { get; }

    /// <summary>
    /// Gets the number of temporary variables used by this method.
    /// </summary>
    int TemporaryCount { get; }

    /// <summary>
    /// Gets the number of arguments expected by this method.
    /// </summary>
    int ArgumentCount { get; }

    /// <summary>
    /// Gets the primitive index if this is a primitive method, or 0 if not.
    /// </summary>
    int PrimitiveIndex { get; }

    /// <summary>
    /// Gets the method header that encodes temporary count, argument count, etc.
    /// </summary>
    int Header { get; }

    /// <summary>
    /// Gets the source code for this method if available.
    /// </summary>
    string? SourceCode { get; }

    /// <summary>
    /// Gets the selector (method name) for this method.

}

/// <summary>
/// Represents a Smalltalk symbol (interned string).

{
    /// <summary>
    /// Gets the string value of this symbol.
    /// </summary>

    /// <summary>
    /// Gets the hash code for this symbol.
    /// </summary>

}

/// <summary>
/// Represents a method dictionary mapping selectors to compiled methods.
/// </summary>
public interface IMethodDictionary : ISmalltalkObject
{
    /// <summary>
    /// Gets a method by selector.
    /// </summary>
    ICompiledMethod? GetMethod(ISymbol selector);

    /// <summary>
    /// Adds or updates a method.
    /// </summary>
    void SetMethod(ISymbol selector, ICompiledMethod method);

    /// <summary>
    /// Removes a method.
    /// </summary>
    bool RemoveMethod(ISymbol selector);

    /// <summary>
    /// Gets all selectors in this dictionary.
    /// </summary>
    IEnumerable<ISymbol> Selectors { get; }
}

// file: Core/Smalltalk80.Core/IVirtualMachine.cs
namespace Smalltalk80.Core;

/// <summary>
/// Defines the contract for a Smalltalk-80 virtual machine implementation.
/// The VM manages object memory, class registry, primitive operations, and bytecode execution.
/// </summary>
public interface IVirtualMachine
{
    /// <summary>
    /// Gets the object memory manager responsible for object allocation and garbage collection.
    /// </summary>
    IObjectMemory ObjectMemory { get; }

    /// <summary>
    /// Gets the class registry that manages all classes and metaclasses in the system.
    /// </summary>
    IClassRegistry ClassRegistry { get; }

    /// <summary>
    /// Gets the primitive operation dispatcher for handling primitive method calls.
    /// </summary>
    IPrimitiveDispatcher PrimitiveDispatcher { get; }

    /// <summary>
    /// Gets the bytecode interpreter for executing compiled methods.
    /// </summary>
    IInterpreter Interpreter { get; }

    /// <summary>
    /// Bootstraps the VM with the minimal class hierarchy and essential methods.
    /// </summary>
    /// <param name="provider">The bootstrap data provider (e.g., JSON file reader).</param>
    /// <returns>A task that completes when bootstrap is finished.</returns>
    Task BootstrapAsync(IBootstrapProvider provider);

    /// <summary>
    /// Executes a compiled method with the given receiver and arguments.
    /// </summary>
    /// <param name="method">The compiled method to execute.</param>
    /// <param name="receiver">The object receiving the message.</param>
    /// <param name="arguments">The method arguments.</param>
    /// <returns>The result of method execution.</returns>
    ISmalltalkObject? Execute(ICompiledMethod method, ISmalltalkObject receiver, params ISmalltalkObject[] arguments);

    /// <summary>
    /// Sends a message to an object and returns the result.
    /// </summary>
    /// <param name="selector">The message selector (method name).</param>
    /// <param name="receiver">The object receiving the message.</param>
    /// <param name="arguments">The message arguments.</param>
    /// <returns>The result of the message send.</returns>
    ISmalltalkObject? SendMessage(ISymbol selector, ISmalltalkObject receiver, params ISmalltalkObject[] arguments);

    /// <summary>
    /// Loads and executes a Smalltalk source file.
    /// </summary>
    /// <param name="filePath">Path to the Smalltalk source file.</param>
    /// <returns>A task that completes when the file is loaded.</returns>
    Task LoadFileAsync(string filePath);

    /// <summary>
    /// Gets a global variable by name.
    /// </summary>
    /// <param name="name">The name of the global variable.</param>
    /// <returns>The global variable value, or null if not found.</returns>
    ISmalltalkObject? GetGlobal(string name);

    /// <summary>
    /// Sets a global variable.
    /// </summary>
    /// <param name="name">The name of the global variable.</param>
    /// <param name="value">The value to set.</param>
    void SetGlobal(string name, ISmalltalkObject value);

    /// <summary>
    /// Gets the current VM configuration.
    /// </summary>
    IVMConfiguration Configuration { get; }

    /// <summary>
    /// Event raised when the VM state changes (e.g., during bootstrap phases).
    /// </summary>
    event EventHandler<VMStateChangedEventArgs>? StateChanged;
}

/// <summary>
/// Represents the VM configuration options.
/// </summary>
public interface IVMConfiguration
{
    /// <summary>
    /// Gets or sets whether to enable tracing for debugging.
    /// </summary>
    bool EnableTracing { get; set; }

    /// <summary>
    /// Gets or sets the maximum stack depth for method calls.
    /// </summary>
    int MaxStackDepth { get; set; }

    /// <summary>
    /// Gets or sets the initial heap size in bytes.
    /// </summary>
    int InitialHeapSize { get; set; }

    /// <summary>
    /// Gets or sets whether to enable the method cache for performance.
    /// </summary>
    bool EnableMethodCache { get; set; }

    /// <summary>
    /// Gets or sets the bootstrap mode (Phase1Only, Phase1And2, Full).
    /// </summary>
    BootstrapMode BootstrapMode { get; set; }
}

/// <summary>
/// Defines the bootstrap modes available for VM initialization.
/// </summary>
public enum BootstrapMode
{
    /// <summary>
    /// Only execute Phase 1 (VM primitives) bootstrap.
    /// </summary>
    Phase1Only,

    /// <summary>
    /// Execute Phase 1 and Phase 2 (messaging) bootstrap.
    /// </summary>
    Phase1And2,

    /// <summary>
    /// Execute all three bootstrap phases including file loading.
    /// </summary>
    Full
}

/// <summary>
/// Event arguments for VM state change notifications.
/// </summary>
public class VMStateChangedEventArgs : EventArgs
{
    /// <summary>
    /// Gets the previous VM state.
    /// </summary>
    public VMState OldState { get; }

    /// <summary>
    /// Gets the new VM state.
    /// </summary>
    public VMState NewState { get; }

    /// <summary>
    /// Gets an optional message describing the state change.
    /// </summary>
    public string? Message { get; }

    /// <summary>
    /// Initializes a new instance of the VMStateChangedEventArgs class.
    /// </summary>
    public VMStateChangedEventArgs(VMState oldState, VMState newState, string? message = null)
    {
        OldState = oldState;
        NewState = newState;
        Message = message;
    }
}

/// <summary>
/// Represents the current state of the virtual machine.
/// </summary>
public enum VMState
{
    /// <summary>
    /// VM has not been initialized.
    /// </summary>
    Uninitialized,

    /// <summary>
    /// VM is executing Phase 1 bootstrap (VM primitives).
    /// </summary>
    BootstrappingPhase1,

    /// <summary>
    /// VM is executing Phase 2 bootstrap (messaging).
    /// </summary>
    BootstrappingPhase2,

    /// <summary>
    /// VM is executing Phase 3 bootstrap (file loading).
    /// </summary>
    BootstrappingPhase3,

    /// <summary>
    /// VM is ready for normal operation.
    /// </summary>
    Ready,

    /// <summary>
    /// VM is executing code.
    /// </summary>
