using System;
using System.Collections.Generic;

namespace Smalltalk80;

/// <summary>
/// Implements basic input/output primitives used by the interpreter.
/// </summary>
public static class InputOutputPrimitives
{
    private static Semaphore? inputSemaphore;
    private static readonly Queue<int> inputBuffer = new();
    private static int sampleInterval;
    private static bool cursorLinked;
    private static (int X, int Y) cursorLocation;
    private static SmalltalkObject? displayScreen;
    private static SmalltalkObject? cursor;

    /// <summary>
    /// Returns a point describing the current cursor location.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    /// <returns><c>true</c> if successful.</returns>
    public static bool MousePoint(Context ctx)
    {
        _ = ctx.Stack.Pop();
        var point = new SmalltalkObject(null, 2);
        point.Fields[0] = cursorLocation.X;
        point.Fields[1] = cursorLocation.Y;
        ctx.Stack.Push(point);
        return true;
    }

    /// <summary>
    /// Sets the cursor location.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool CursorLocPut(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject point)
            return false;
        _ = ctx.Stack.Pop();
        if (point.Fields.Length < 2)
            return false;
        cursorLocation = (Convert.ToInt32(point.Fields[0]), Convert.ToInt32(point.Fields[1]));
        ctx.Stack.Push(point);
        return true;
    }

    /// <summary>
    /// Links or unlinks the cursor from the pointing device.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool CursorLink(Context ctx)
    {
        if (ctx.Stack.Pop() is not bool link)
            return false;
        _ = ctx.Stack.Pop();
        cursorLinked = link;
        ctx.Stack.Push(link);
        return true;
    }

    /// <summary>
    /// Sets the <see cref="Semaphore"/> used to signal input events.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool InputSemaphore(Context ctx)
    {
        if (ctx.Stack.Pop() is not Semaphore sem)
            return false;
        _ = ctx.Stack.Pop();
        inputSemaphore = sem;
        ctx.Stack.Push(sem);
        return true;
    }

    /// Sets the sample interval for pointing device movement.

        if (ctx.Stack.Pop() is not int interval)

        _ = ctx.Stack.Pop();

        return true;

        _ = ctx.Stack.Pop();

            return false;

        ctx.Stack.Push(word);
        return true;
    }

    /// <summary>
    /// Stores the current time in seconds since 1901 into a byte array.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool TimeWordsInto(Context ctx)
    {
        if (ctx.Stack.Pop() is not SmalltalkObject bytes || bytes.Fields.Length < 4)
            return false;
        _ = ctx.Stack.Pop();
        var seconds = (int)(DateTime.UtcNow - new DateTime(1901, 1, 1)).TotalSeconds;
        for (int i = 0; i < 4; i++)
            bytes.Fields[i] = (byte)((seconds >> (8 * i)) & 0xFF);
        ctx.Stack.Push(bytes);
        return true;
    }

    /// Stores the current tick count in a byte array.
    /// </summary>
    /// <param name="ctx">The active context.</param>

    {
        if (ctx.Stack.Pop() is not SmalltalkObject bytes || bytes.Fields.Length < 4)
            return false;
        _ = ctx.Stack.Pop();
        int ticks = Environment.TickCount;
        for (int i = 0; i < 4; i++)
            bytes.Fields[i] = (byte)((ticks >> (8 * i)) & 0xFF);
        ctx.Stack.Push(bytes);
        return true;
    }

    /// <summary>
    /// Schedules a semaphore to be signaled at the specified tick count. This implementation uses a timer.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool SignalAtTick(Context ctx)
    {
        if (ctx.Stack.Pop() is not int tick || ctx.Stack.Pop() is not Semaphore sem)
            return false;
        _ = ctx.Stack.Pop();
        var delay = Math.Max(0, tick - Environment.TickCount);
        var timer = new System.Threading.Timer(_ => inputSemaphore?.Waiting.Enqueue(new Process(ctx)), null, delay, System.Threading.Timeout.Infinite);
        ctx.Stack.Push(sem);
        return true;
    }

    /// <param name="ctx">The active context.</param>
    public static bool DrawLoop(Context ctx)
    {
        _ = ctx.Stack.Pop();

    }

    /// <summary>
    /// Replaces bytes in a <see cref="SmalltalkObject"/> array with bytes from another array.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool StringReplace(Context ctx)
    {
        if (ctx.Stack.Pop() is not int start ||
            ctx.Stack.Pop() is not SmalltalkObject source ||
            ctx.Stack.Pop() is not int stop ||
            ctx.Stack.Pop() is not SmalltalkObject target)
            return false;
        _ = ctx.Stack.Pop();
        int count = Math.Min(stop - 1, target.Fields.Length) - Math.Min(start - 1, target.Fields.Length) + 1;
        if (count <= 0)
            return false;
        for (int i = 0; i < count && start - 1 + i < target.Fields.Length && i < source.Fields.Length; i++)
            target.Fields[start - 1 + i] = source.Fields[i];
        ctx.Stack.Push(target);
        return true;
    }
}

    /// <summary>
    /// Returns an approximation of free object slots in the object memory.
    /// </summary>

    public static bool CoreLeft(Context ctx)
    {
        _ = ctx.Stack.Pop();
        int free = int.MaxValue - ObjectMemory.Objects.Count;
        ctx.Stack.Push(free);
        return true;
    }

    /// <summary>
    /// Requests termination of the process. This implementation does not exit during testing.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool Quit(Context ctx)
    {
        _ = ctx.Stack.Pop();
        ctx.Stack.Push(null);
        return true;
    }

    /// <summary>
    /// Breaks into the debugger if attached.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool ExitToDebugger(Context ctx)
    {
        _ = ctx.Stack.Pop();
        System.Diagnostics.Debugger.Launch();
        ctx.Stack.Push(null);
        return true;
    }

    /// <summary>
    /// Returns a placeholder value representing remaining oops.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool OopsLeft(Context ctx)
    {
        _ = ctx.Stack.Pop();
        ctx.Stack.Push(int.MaxValue - ObjectMemory.Objects.Count);
        return true;
    }

    /// <summary>
    /// Signals a semaphore when the number of allocated objects reaches a threshold.
    /// </summary>
    /// <param name="ctx">The active context.</param>
    public static bool SignalAtOopsLeftWordsLeft(Context ctx)
    {
        if (ctx.Stack.Pop() is not int threshold || ctx.Stack.Pop() is not Semaphore sem)
            return false;
        _ = ctx.Stack.Pop();
        if (ObjectMemory.Objects.Count >= threshold)
        {
            ProcessorScheduler.Instance.Resume(new Process(ctx));
        }
        ctx.Stack.Push(sem);
        return true;
    }
}

using System;
using System.Collections.Generic;
using System.Linq;

/// <summary>
/// Implements a real object memory system following the Smalltalk-80 specification.
/// Reference: Virtual Machine - Object Memory.md, lines 1-202
/// </summary>
public class RealObjectMemory
{
    /// <summary>
    /// The heap where all objects are stored.
    /// </summary>
    private readonly List<HeapObject> heap = new();

    /// <summary>
    /// Free object table for garbage collection.
    /// </summary>
    private readonly Stack<int> freeObjects = new();

    /// <summary>
    /// Reference counting table for automatic memory management.
    /// </summary>
    private readonly Dictionary<int, int> referenceCounts = new();

    /// <summary>
    /// Creates a new object with the specified class and number of fields.
    /// </summary>
    /// <param name="classPointer">The class of the new object.</param>
    /// <param name="fieldCount">Number of instance variables.</param>
    /// <returns>Object pointer to the new object.</returns>
    public int CreateObject(SmalltalkObject classPointer, int fieldCount)
    {
        int objectPointer;

        if (freeObjects.Count > 0)
        {
            objectPointer = freeObjects.Pop();
            heap[objectPointer] = new HeapObject
            {
                ClassPointer = classPointer,
                Fields = new object?[fieldCount],
                Size = fieldCount + 2 // Header size
            };
        }
        else
        {
            objectPointer = heap.Count;
            heap.Add(new HeapObject
            {
                ClassPointer = classPointer,
                Fields = new object?[fieldCount],
                Size = fieldCount + 2
            });
        }

        referenceCounts[objectPointer] = 1;
        return objectPointer;
    }

    /// <summary>
    /// Retrieves the object at the specified pointer.
    /// </summary>
    /// <param name="objectPointer">The object pointer.</param>
    /// <returns>The heap object.</returns>
    public HeapObject GetObject(int objectPointer)
    {
        if (objectPointer < 0 || objectPointer >= heap.Count)
            throw new ArgumentOutOfRangeException(nameof(objectPointer));

        if (obj.Fields[fieldIndex] is int oldPointer)

        }

            }
        }

        // Clear the object
        heap[objectPointer] = new HeapObject();

        referenceCounts.Remove(objectPointer);

    /// Performs garbage collection using mark and sweep algorithm.
    /// </summary>
    public void GarbageCollect()
    {
        // Mark phase: mark all reachable objects
        var marked = new HashSet<int>();
        var roots = GetRootObjects();

        foreach (var root in roots)
        {
            MarkObject(root, marked);
        }

        // Sweep phase: deallocate unmarked objects
        for (int i = 0; i < heap.Count; i++)
        {
            if (!marked.Contains(i) && referenceCounts.ContainsKey(i))
            {
                DeallocateObject(i);
            }
        }
    }

        // Format: [superclass, methodDict, format, ...]
        const int METHOD_DICT_INDEX = 1;
        
        if (classObj.Fields == null || classObj.Fields.Length <= METHOD_DICT_INDEX)
            return null;
            
        var methodDictObj = classObj.Fields[METHOD_DICT_INDEX];
        if (methodDictObj is not SmalltalkObject methodDict)

        // MethodDictionary format:
        // Field 0: tally (number of entries)
        // Fields 1..n/2: keys (selectors)
        // Fields n/2+1..n: values (compiled methods)
        
        if (methodDict.Fields == null || methodDict.Fields.Length < 2)
            return null;
            
        var tally = methodDict.Fields[0] as int? ?? 0;
        if (tally == 0)
            return null;
            
        // The dictionary is split in half - first half is keys, second half is values
        var halfSize = (methodDict.Fields.Length - 1) / 2;
        var keysStart = 1;
        var valuesStart = keysStart + halfSize;
        
        // Search for the selector in the keys
        for (int i = 0; i < halfSize; i++)
        {

            if (key is Symbol keySymbol && keySymbol.Equals(selector))
            {

                var value = methodDict.Fields[valuesStart + i];

                    return compiledMethod;

    /// </summary>

    /// <summary>
    /// Determines if a SmalltalkObject is likely a class object based on its structure.
    /// </summary>
    private bool IsLikelyClassObject(SmalltalkObject obj)
    {
        // Class objects typically have specific characteristics:
        // 1. They often have a ClassPointer that points to a metaclass
        // 2. They may have fields that reference method dictionaries, superclasses, etc.
        // 3. The structure varies, but we can make educated guesses

            return false;
            
        // Check if this object's class name contains "class" or "Class" 
        // (this is a heuristic based on common Smalltalk patterns)
        var className = GetObjectClassName(obj);
        if (className?.ToLower().Contains("class") == true)
            return true;
            
        // Check if the object has fields that might indicate it's a class
        // (method dictionary, superclass reference, instance variables, etc.)
        if (obj.Fields.Length >= 3) // Classes typically have multiple fields
        {
            // Look for patterns in field contents that suggest this is a class
            foreach (var field in obj.Fields)

            // Special handling for Symbol and String objects to show their value
            if (fieldClassName == "Symbol" || fieldClassName == "String")
            {
                // Try to get the value from the first field
                if (fieldObj.Fields.Length > 0 && fieldObj.Fields[0] is string value)
                {
                    if (fieldClassName == "Symbol")
                        return $"#{value}";
                    else
                        return $"\"{value}\"";
                }
            }
            
            // Check if this object itself is registered as a class
            // Note: Without the ability to enumerate globals, we can't find the name
            // This is a limitation of the new API
            
            // For non-global objects, show them more clearly
            // Use "a/an" article for instances
            string article = "aeiouAEIOU".Contains(fieldClassName[0]) ? "an" : "a";
            return $"{article} {fieldClassName} @ #{fieldObj.Oop}";
        }
        
        if (field is string str)
            return $"String: \"{str}\"";
            
        if (field is int intVal)
            return $"Integer: {intVal}";
            
        if (field is bool boolVal)
            return $"Boolean: {boolVal}";
            
        return $"{field.GetType().Name}: {field}";

    /// <summary>
    /// Dumps object relationships to show the object graph structure.
    /// </summary>
    private void DumpObjectRelationships(List<SmalltalkObject> allObjects)
    {
        Console.WriteLine("🔗 Object Relationships:");
        
        if (allObjects.Count == 0)
        {
            Console.WriteLine("  No objects to analyze");
            return;
        }
        
        // Count references between objects
        var referenceCount = new Dictionary<int, int>(); // OOP -> count of references to it
        var outgoingRefs = new Dictionary<int, List<int>>(); // OOP -> list of OOPs it references
        
        foreach (var obj in allObjects)
        {
            referenceCount[obj.Oop] = 0;
            outgoingRefs[obj.Oop] = new List<int>();
            
            // Examine each field for references to other objects
            foreach (var field in obj.Fields)
            {
                if (field is SmalltalkObject referencedObj)
                {
                    referenceCount[referencedObj.Oop] = referenceCount.GetValueOrDefault(referencedObj.Oop, 0) + 1;
                    outgoingRefs[obj.Oop].Add(referencedObj.Oop);
                }
            }
        }
        
        // Show objects with high reference counts (likely important objects)
        var highlyReferenced = referenceCount
            .Where(kvp => kvp.Value > 0)
            .OrderByDescending(kvp => kvp.Value)
            .Take(5)
            .ToList();
            
        if (highlyReferenced.Count > 0)
        {
            Console.WriteLine("  📈 Most Referenced Objects:");
            foreach (var (oop, refCount) in highlyReferenced)
            {
                var obj = ObjectMemory.FromOop(oop);
                var className = obj != null ? GetObjectClassName(obj) ?? "(unknown)" : "(invalid)";
                Console.WriteLine($"    #{oop} ({className}): {refCount} references");
            }
        }
        
        // Show objects that reference many others (likely container objects)
        var highConnectivity = outgoingRefs
            .Where(kvp => kvp.Value.Count > 0)
            .OrderByDescending(kvp => kvp.Value.Count)
            .Take(3)
            .ToList();
            
        if (highConnectivity.Count > 0)
        {
            Console.WriteLine("  📤 Objects with Most Outgoing References:");
            foreach (var (oop, refs) in highConnectivity)
            {
                var obj = ObjectMemory.FromOop(oop);
                var className = obj != null ? GetObjectClassName(obj) ?? "(unknown)" : "(invalid)";
                Console.WriteLine($"    #{oop} ({className}): references {refs.Count} objects");
            }
        }
        
        Console.WriteLine($"  📊 Total objects: {allObjects.Count}, Total references: {referenceCount.Values.Sum()}");
    }

// file: TestSmalltalkGrammar/Program.cs
using System;
using System.IO;
using Antlr4.Runtime;
using Antlr4.Runtime.Misc;
using Smalltalk80.Generated;

namespace Smalltalk80.Tools
{
    /// <summary>
    /// Utility program that validates the Smalltalk-80 grammar by parsing
    /// a classic <c>file-out</c> source file and reporting any syntax errors.
    /// </summary>
    public static class TestSmalltalkGrammar
    {
        /// <summary>
        /// Application entry point.
        /// </summary>
        /// <param name="args">An array containing the path to the file-out source.</param>
        /// <returns>Zero if parsing succeeds; otherwise one.</returns>
        public static int Main(string[] args)
        {
            if (args.Length != 1)
            {
                Console.Error.WriteLine("Usage: TestSmalltalkGrammar <path to Smalltalk-80.st>");
                return 1;
            }

            // 1) Read the entire file-out source
            var source = File.ReadAllText(args[0]);

            // 2) Run it through the ANTLR4 pipeline
            var input = CharStreams.fromString(source);
            var lexer = new Smalltalk80Lexer(input);
            var tokens = new CommonTokenStream(lexer);

            // Dump tokens for debugging
            tokens.Fill(); // Load all tokens

            Console.WriteLine("🔍 Token Stream:");
            foreach (var token in tokens.GetTokens())
            {
                var tokenName = lexer.Vocabulary.GetSymbolicName(token.Type);
                Console.WriteLine($"  [{token.Line}:{token.Column}] {tokenName} → '{token.Text}'");
            }
            Console.WriteLine();

            var parser = new Smalltalk80Parser(tokens);
            parser.Interpreter.PredictionMode = Antlr4.Runtime.Atn.PredictionMode.LL_EXACT_AMBIG_DETECTION;
            parser.Trace = true;

            // 3) Hook up a listener to catch any lex/parse errors
            parser.RemoveErrorListeners();
            var listener = new CountingErrorListener();
            parser.AddErrorListener(listener);

            // 4) Parse in "fileIn" (batch) mode
            parser.fileIn();

            // 5) Report results
            if (listener.ErrorCount == 0 && parser.NumberOfSyntaxErrors == 0)
            {
                Console.WriteLine("🎉 Parsed entire file-out with zero syntax errors.");
                return 0;
            }
            else
            {
                var total = parser.NumberOfSyntaxErrors + listener.ErrorCount;
                Console.Error.WriteLine($"❌ Found {total} syntax errors.");
                return 1;
            }
        }

        /// <summary>
        /// Error listener that counts syntax errors encountered during parsing.
        /// </summary>
        private sealed class CountingErrorListener : BaseErrorListener
        {
            /// <summary>
            /// Gets the number of syntax errors recorded.
            /// </summary>
            public int ErrorCount { get; private set; }

            /// <inheritdoc />
            public override void SyntaxError(
                TextWriter output,
                IRecognizer recognizer,
                IToken offendingSymbol,
                int line,
                int charPositionInLine,
                string msg,
                RecognitionException e)
            {
                ErrorCount++;
                Console.Error.WriteLine($"Syntax Error @ line {line}:{charPositionInLine} – {msg}");
                Environment.Exit(1); // Immediately stop the program
            }
        }
    }
}
