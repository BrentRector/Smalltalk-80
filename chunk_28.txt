using System;
using Xunit;
using Xunit.Abstractions;

namespace Smalltalk80.Tests;

/// <summary>
/// Simplified execution tests to validate basic method compilation and execution 
/// without complex file-out format dependencies.
/// </summary>
public class SimpleExecutionTests(ITestOutputHelper output)
{
    [Fact]
    public void SimpleMethod_ReturnConstant_ExecutesCorrectly()
    {
        // Reset system 
        ObjectMemory.Reset();

        // Create a simple method that returns a constant
        var vm = new VirtualMachine();
        var tempFile = CreateTempMethodFile("Object>>testMethod", "^42!");

        vm.LoadSourceFile(tempFile);

        // Verify method was loaded
        var objectClass = SystemDictionary.GetClass("Object");
        var method = objectClass.Lookup(new Symbol("testMethod"));
        Assert.NotNull(method);

        // Execute the method
        var objectInstance = new SmalltalkObject(objectClass, 0);
        var interpreter = new Interpreter();
        interpreter.SetActiveContext(objectInstance, method);
        var result = interpreter.ExecuteMethod();

        Assert.Equal(42, result);
        output.WriteLine("✅ Simple method execution test passed");

        CleanupTempFile(tempFile);
    }

    [Fact]
    public void SimpleMethod_WithArgument_ExecutesCorrectly()
    {
        // Reset system 
        ObjectMemory.Reset();

        // Create a method that adds 1 to its argument
        var vm = new VirtualMachine();
        var tempFile = CreateTempMethodFile("Object>>addOne: x", "^x + 1!");

        vm.LoadSourceFile(tempFile);

        // Verify method was loaded
        var objectClass = SystemDictionary.GetClass("Object");
        var method = objectClass.Lookup(new Symbol("addOne:"));
        Assert.NotNull(method);

        // Execute the method with argument
        var objectInstance = new SmalltalkObject(objectClass, 0);
        var interpreter = new Interpreter();
        interpreter.SetActiveContext(objectInstance, method, 5);
        var result = interpreter.ExecuteMethod();

        Assert.Equal(6, result);
        output.WriteLine("✅ Method with argument test passed");

        CleanupTempFile(tempFile);
    }

    [Fact]
    public void SimpleMethod_MultipleArithmetic_ExecutesCorrectly()
    {
        // Reset system
        ObjectMemory.Reset();

        // Create a method that does multiple arithmetic operations
        var vm = new VirtualMachine();
        var tempFile = CreateTempMethodFile("Object>>calculate: a with: b", "^(a + b) * 2!");

        vm.LoadSourceFile(tempFile);

        // Execute the method
        var objectClass = SystemDictionary.GetClass("Object");
        var method = objectClass.Lookup(new Symbol("calculate:with:"));
        Assert.NotNull(method);

        var objectInstance = new SmalltalkObject(objectClass, 0);
        var interpreter = new Interpreter();
        interpreter.SetActiveContext(objectInstance, method, 3, 7);
        var result = interpreter.ExecuteMethod();

        Assert.Equal(20, result); // (3 + 7) * 2 = 20
        output.WriteLine("✅ Multiple arithmetic test passed");

        CleanupTempFile(tempFile);
    }

    /// <summary>
    /// Creates a temporary file with the specified method definition.
    /// </summary>
    private static string CreateTempMethodFile(string header, string body)
    {
        var tempFile = System.IO.Path.GetTempFileName();
        var content = $"{header}\n{body}";
        System.IO.File.WriteAllText(tempFile, content);
        return tempFile;
    }

    /// <summary>
    /// Cleans up the temporary file.
    /// </summary>
    private static void CleanupTempFile(string tempFile)
    {
        if (System.IO.File.Exists(tempFile))
        {
            System.IO.File.Delete(tempFile);
        }
    }
}

using System;
using Xunit;
using Xunit.Abstractions;

namespace Smalltalk80.Tests;

public class SimpleMethodParsingTest(ITestOutputHelper output)
{
    [Fact]
    public void ComplexKeywordMethod_ParsesCorrectly()
    {
        ObjectMemory.Reset();

        var fileOutSource = @"
Object subclass: #Counter
    instanceVariableNames: 'count'
    classVariableNames: ''
    poolDictionaries: ''
    category: 'Test-Debug'!

!Counter methodsFor: 'incrementing'!
incrementBy: amount
    count _ count + amount.
    ^self
! !
";

        var tempFile = CreateTempFile(fileOutSource);

        try
        {
            var vm = new VirtualMachine();
            vm.LoadSystemFile(tempFile);

            var testClass = SystemDictionary.GetClass("Counter");
            Assert.NotNull(testClass);

            output.WriteLine("=== Methods loaded ===");
            foreach (var kvp in testClass.MethodDictionary)
            {
                output.WriteLine($"  {kvp.Key.Name}");
            }

            // This should be 'incrementBy:' not 'incrementBy:count:'
            var method = testClass.Lookup(new Symbol("incrementBy:"));
            output.WriteLine($"Method 'incrementBy:' found: {method != null}");

            var wrongMethod = testClass.Lookup(new Symbol("incrementBy:count:"));
            output.WriteLine($"Method 'incrementBy:count:' found: {wrongMethod != null}");

            Assert.NotNull(method);
            Assert.Null(wrongMethod);

            output.WriteLine("✅ Complex keyword method parsed correctly");
        }
        finally
        {
            System.IO.File.Delete(tempFile);
        }
    }

    private static string CreateTempFile(string content)
    {
        var tempFile = System.IO.Path.GetTempFileName();
        System.IO.File.WriteAllText(tempFile, content);
        return tempFile;
    }
}

                    (byte)Bytecode.ReturnTopFromMethod

                Literals = new object[] { new Symbol("foo") },

            };

            var result = vm.Interpreter.Evaluate(callMethod, instance);
            Assert.Equal(3, result);
        }
        finally
        {
            File.Delete(tempPath);
        }
    }

    private static FileNode ParseFileIn(string source)
    {
        var reader = new StringReader(source);
        var input = new Antlr4.Runtime.AntlrInputStream(reader);
        var lexer = new Smalltalk80.Generated.Smalltalk80Lexer(input);
        var tokens = new Antlr4.Runtime.CommonTokenStream(lexer);
        var parser = new Smalltalk80.Generated.Smalltalk80Parser(tokens);
        parser.RemoveErrorListeners();

        var fileCtx = parser.fileIn();
        if (parser.NumberOfSyntaxErrors > 0)
            throw new InvalidOperationException("Parse errors detected");

        var visitor = new ClassDefBuildingVisitor();
        return (FileNode)visitor.VisitFileIn(fileCtx)!;
    }

    private static void AssertClassExists(string className, string? expectedSuperclass = null)
    {
        Assert.True(SystemDictionary.TryGetClass(className, out var cls), $"Class '{className}' should exist");

        {
            Assert.True(SystemDictionary.TryGetClass(expectedSuperclass, out var super), $"Superclass '{expectedSuperclass}' should exist");
            Assert.Equal(super, cls!.Superclass);
        }
    }

    private static void AssertClassHasInstanceVariables(string className, params string[] expectedVars)

        var cls = SystemDictionary.GetClass(className);

            Assert.Contains(expectedVar, cls.InstanceVariables);

        var tokens = new Antlr4.Runtime.CommonTokenStream(lexer);
        var parser = new Smalltalk80.Generated.Smalltalk80Parser(tokens);
        parser.RemoveErrorListeners();

        var fileCtx = parser.fileIn();
        if (parser.NumberOfSyntaxErrors > 0)
            throw new InvalidOperationException("Parse errors detected");

        var visitor = new ClassDefBuildingVisitor();
        return (FileNode)visitor.VisitFileIn(fileCtx)!;

{
    private readonly Dictionary<string, int> temporaryGlobals = new();
    private readonly Dictionary<string, int> temporaryClasses = new();
    private readonly Dictionary<string, int> temporarySymbols = new();
    private bool isActive = true;

    /// Check if the bootstrap context is still active.
    /// Once deactivated, all operations should fail.
    /// </summary>

    /// Register a temporary global during bootstrap.
    /// This should ONLY be called during Phase 1.
    /// </summary>

    {
        if (!isActive)
            throw new InvalidOperationException("Bootstrap context has been deactivated. Use Smalltalk dictionary instead.");

        }

    /// <summary>
    /// Execute a class definition via bytecode
    /// </summary>
    private void ExecuteClassDefinition(ClassDefinitionNode classDef)
    {
        Console.WriteLine($"Creating class {classDef.ClassName} via bytecode...");
        
        // Generate bytecode for:
        // SuperClass subclass: #ClassName
        //     instanceVariableNames: 'var1 var2'
        //     classVariableNames: 'ClassVar1'
        //     poolDictionaries: ''
        //     category: 'Category'
        
        var bytecode = new BytecodeBuilder();
        
        // Push superclass
        if (classDef.Superclass == "nil" || classDef.Superclass == null)
        {
            bytecode.PushNil();
        }
        else
        {
            bytecode.PushGlobal(classDef.Superclass);
        }
        
        // Push class name symbol
        bytecode.PushLiteral(vm.PrimitiveInternSymbol(classDef.ClassName));
        
        // Push instance variables string
        bytecode.PushLiteral(vm.PrimitiveInternString(
            string.Join(" ", classDef.InstanceVariables)));
        
        // Push class variables string
        bytecode.PushLiteral(vm.PrimitiveInternString(
            string.Join(" ", classDef.ClassVariables)));
        
        // Push pool dictionaries string
        bytecode.PushLiteral(vm.PrimitiveInternString(
            string.Join(" ", classDef.PoolDictionaries)));
        
        // Push category string
        bytecode.PushLiteral(vm.PrimitiveInternString(
            classDef.Category ?? "Unclassified"));
        
        // Send subclass:instanceVariableNames:classVariableNames:poolDictionaries:category:
        bytecode.SendSelector("subclass:instanceVariableNames:classVariableNames:poolDictionaries:category:", 5);
        
        // Store result in global
        bytecode.StoreGlobal(classDef.ClassName);
        
        // Execute the bytecode
        vm.ExecuteBytecode(vm.PrimitiveLookupGlobal("Smalltalk"), bytecode.GetBytes(), new int[0]);
        
        // Track the new class
        globalOops[classDef.ClassName] = vm.PrimitiveLookupClass(classDef.ClassName);
    }
    
    /// <summary>
    /// Execute method definitions via bytecode
    /// </summary>
    private void ExecuteMethodCategory(MethodCategoryNode category)
    {
        var className = category.Header.ClassName;
        var isClassSide = category.Header.IsClassSide;
        
        Console.WriteLine($"Installing methods in {className}{(isClassSide ? " class" : "")}...");
        
        foreach (var method in category.Methods)

            ExecuteMethodInstallation(className, method, isClassSide);

    /// <summary>
    /// Install a single method via bytecode
    /// </summary>
    private void ExecuteMethodInstallation(string className, MethodDefinitionNode method, bool isClassSide)
    {
        var bytecode = new BytecodeBuilder();
        
        // Push the class (or metaclass if class-side)
        if (isClassSide)
        {
            bytecode.PushGlobal(className);
            bytecode.SendSelector("class", 0); // Get metaclass
        }
        else
        {
            bytecode.PushGlobal(className);
        }
        
        // Push the method source
        bytecode.PushLiteral(vm.PrimitiveInternString(method.Source));
        
        // Send compile:
        bytecode.SendSelector("compile:", 1);
        
        // Pop result
        bytecode.Pop();
        
        // Execute
        vm.ExecuteBytecode(vm.PrimitiveLookupGlobal("Smalltalk"), bytecode.GetBytes(), new int[0]);
    }

            case VariableExpression variable:

                break;
                
            case LiteralExpression literal:

                    bytecode.PushLiteral(vm.PrimitiveInternString(str));
                }
                else if (literal.Value is int num)
                {
                    bytecode.PushLiteral(num);

    /// <summary>
    /// Helper class to build bytecode sequences
    /// </summary>
    private class BytecodeBuilder
    {
        private readonly List<byte> bytes = new();
        private readonly List<object> literals = new();
        
        public void PushReceiver() => bytes.Add(PUSH_RECEIVER);
        public void PushTrue() => bytes.Add(PUSH_TRUE);
        public void PushFalse() => bytes.Add(PUSH_FALSE);
        public void PushNil() => bytes.Add(PUSH_NIL);
        public void Dup() => bytes.Add(DUP);
        public void Pop() => bytes.Add(POP);
        public void ReturnTop() => bytes.Add(RETURN_TOP);
        public void ReturnReceiver() => bytes.Add(RETURN_RECEIVER);
        
        public void PushLiteral(object literal)
        {
            var index = literals.Count;
            literals.Add(literal);
            bytes.Add((byte)(PUSH_LITERAL + index));
        }
        
        public void PushGlobal(string globalName)
        {
            bytes.Add(PUSH_GLOBAL);
            bytes.AddRange(Encoding.UTF8.GetBytes(globalName));
            bytes.Add(0); // null terminator
        }
        
        public void StoreGlobal(string globalName)
        {
            // Custom bytecode for storing global
            bytes.Add(0x50); // STORE_GLOBAL
            bytes.AddRange(Encoding.UTF8.GetBytes(globalName));
            bytes.Add(0);
        }
        
        public void SendSelector(string selector, int argCount)
        {
            bytes.Add((byte)(SEND_SELECTOR + argCount));
            bytes.AddRange(Encoding.UTF8.GetBytes(selector));
            bytes.Add(0);
        }

        {
            // In a real implementation, would need to encode literals properly
